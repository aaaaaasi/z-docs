   175→function viewFromUrl(): { view: WorkspaceView; docId: string | null; target: AppTarget | null } {
   176→  if (typeof window === "undefined") return { view: "home", docId: null, target: null }
   177→  const p = new URLSearchParams(window.location.search)
   178→  const doc = p.get("doc")
   179→  if (doc) return { view: "editor", docId: doc, target: null }
   180→  const app = p.get("app")
   181→  if (app === "sheets" || app === "slides" || app === "forms") {
   182→    const formModeRaw = p.get("mode")
   183→    const formMode =
   184→      app === "forms" && (formModeRaw === "fill" || formModeRaw === "responses" || formModeRaw === "edit")
   185→        ? (formModeRaw as FormOpenMode)
   186→        : null
   187→    return {
   188→      view: app,
   189→      docId: null,
   190→      target: { app, id: p.get("entity"), formMode },
   191→    }
   192→  }
   193→  if (app === "activity") return { view: "activity", docId: null, target: null }
   194→  if (app === "settings") return { view: "settings", docId: null, target: null }
   195→  return { view: "home", docId: null, target: null }
   196→}

   222→  fetchMe: async () => {
   223→    try {
   224→      const res = await fetch("/api/auth/me", { cache: "no-store" })
   225→      if (res.ok) {
   226→        const data = (await res.json()) as { user: AuthUser }
   227→        set({ authUser: data.user, authLoaded: true, authError: null })
   228→        // keep the presence identity in sync with the real account
   229→        try {
   230→          saveLocalUser({ id: data.user.id, name: data.user.name, color: data.user.color })
   231→        } catch {
   232→          /* identity is best-effort */
   233→        }
   234→      } else {
   235→        set({ authUser: null, authLoaded: true })
   236→      }
   237→    } catch {
   238→      set({ authUser: null, authLoaded: true })
   239→    }
   240→  },

   242→  /** Shared post-auth flow: exit guest mode, upload local-only content to
   243→   *  the cloud account, then reload the (now server-backed) workspace. */
   244→  finishAuth: async (user: AuthUser): Promise<AuthResult> => {
   245→    const wasGuest = get().guestMode
   246→    let snapshot: ReturnType<typeof readGuestSnapshot> | null = null
   247→    if (wasGuest && hasGuestData()) snapshot = readGuestSnapshot()
   248→    // leave local mode BEFORE any data requests so they hit the real server
   249→    uninstallGuestApi()
   250→    try {
   251→      window.localStorage.removeItem(GUEST_FLAG)
   252→    } catch {
   253→      /* storage unavailable */
   254→    }
   255→    set({ authUser: user, guestMode: false, authScreen: false, authBusy: false, authError: null, authLoaded: true })
   256→    try {
   257→      saveLocalUser({ id: user.id, name: user.name, color: user.color })
   258→    } catch {
   259→      /* best-effort */
   260→    }
   261→    let imported = 0
   262→    if (snapshot) {
   263→      try {
   264→        const res = await fetch("/api/import/local", {
   265→          method: "POST",
   266→          headers: { "Content-Type": "application/json" },
   267→          body: JSON.stringify(snapshot),
   268→        })
   269→        if (res.ok) {
   270→          const data = (await res.json()) as { imported?: Record<string, number> }
   271→          imported = Object.values(data.imported ?? {}).reduce((a, b) => a + b, 0)
   272→          clearGuestData()
   273→        }
   274→      } catch {
   275→        /* keep local data — it syncs again on the next sign-in */
   276→      }
   277→    }
   278→    await get().refresh({ silent: true })
   279→    await get().refreshFolders()
   280→    await get().loadTags()
   281→    void get().fetchStorage()
   282→    // local-only entity ids (local-…) no longer resolve in cloud mode —
   283→    // land the freshly signed-in user on a clean home view
   284→    if (typeof window !== "undefined") {
   285→      window.history.replaceState({}, "", "/")
   286→    }
   287→    set({ view: "home", currentDocId: null, appTarget: null })
   288→    return { ok: true, imported }
   289→  },
   290→
   291→  login: async (email, password) => {
   292→    set({ authBusy: true, authError: null })
   293→    try {
   294→      const res = await fetch("/api/auth/login", {

   498→  refresh: async (opts) => {

   502→      const { filter, searchQuery, activeFolderId } = get()
   503→      const params = new URLSearchParams({ filter })

   505→      if (filter === "folder" && activeFolderId && !searchQuery.trim()) {
   506→        params.set("folder", activeFolderId)

   509→      const res = await fetch(`/api/documents?${params.toString()}`)
   510→      if (!res.ok) throw new Error(`Failed to load documents (${res.status})`)
   511→      const data = (await res.json()) as { documents: DocumentMeta[] }
   512→      set({ documents: data.documents ?? [] })

   517→    }

   531→  createDoc: async (opts) => {
   532→    try {
   533→      const template = opts?.templateId ? getTemplate(opts.templateId) : undefined
   534→      // template title/content are localized per the current UI language
   535→      const lang = getCurrentLang()
   536→      const tpl = template ? localizedTemplate(template, lang) : undefined
   537→      let content = opts?.content ?? tpl?.content ?? ""
   538→      // workspace default font (Settings → Workspace defaults) applies to fresh blank docs
   539→      if (!opts?.content && !tpl?.content) {
   540→        try {
   541→          if (typeof window !== "undefined" && window.localStorage.getItem("zdocs-default-font") === "serif") {
   542→            content = '<p style="font-family: Georgia, serif"><br></p>'
   543→          }
   544→        } catch {
   545→          // storage unavailable — keep plain default
   546→        }
   547→      }
   548→      const body = {
   549→        title: opts?.title ?? tpl?.title ?? tForLang(lang, "Untitled document"),
   550→        content,
   551→      }
   552→      const res = await fetch("/api/documents", {
   553→        method: "POST",
   554→        headers: { "Content-Type": "application/json" },
   555→        body: JSON.stringify(body),
   556→      })
   557→      if (!res.ok) throw new Error("Failed to create document")
   558→      const data = (await res.json()) as { document: DocumentDTO }
   559→      const id = data.document.id
   560→      // optionally file the new doc into a folder
   561→      const folderId = opts?.folderId ?? null
   562→      if (folderId) {
   563→        await fetch(`/api/documents/${id}`, {
   564→          method: "PATCH",
   565→          headers: { "Content-Type": "application/json" },
   566→          body: JSON.stringify({ folderId }),
   567→        }).catch(() => {})
   568→      }
   569→      await get().refresh({ silent: true })
   570→      return id
   571→    } catch (e) {
   572→      set({ error: e instanceof Error ? e.message : "Create failed" })
   573→      return null
   574→    }
   575→  },
   576→
   577→  openDoc: (id, opts) => {
   578→    if (typeof window !== "undefined") {
   579→      window.history.pushState({}, "", `/?doc=${encodeURIComponent(id)}`)
   580→    }
   581→    set({ view: "editor", currentDocId: id, openAiOnEditor: !!opts?.ai })
   582→  },
   583→
   584→  goHome: () => {
   585→    if (typeof window !== "undefined") {
   586→      window.history.pushState({}, "", "/")
   587→    }
   588→    set({ view: "home", currentDocId: null, openAiOnEditor: false, appTarget: null })
   589→    void get().refresh({ silent: true })
   590→    void get().refreshFolders()
   591→    void get().loadTags()
   592→  },
   593→
   594→  renameDoc: async (id, title) => {
   595→    const optimistic = get().documents.map((d) => (d.id === id ? { ...d, title } : d))
   596→    set({ documents: optimistic })
   597→    try {
   598→      await fetch(`/api/documents/${id}`, {
   599→        method: "PATCH",
   600→        headers: { "Content-Type": "application/json" },
   601→        body: JSON.stringify({ title }),
   602→      })
   603→    } catch {
   604→      void get().refresh({ silent: true })
   605→    }
   606→  },
   607→
   608→  toggleStar: async (id) => {
   609→    const doc = get().documents.find((d) => d.id === id)
   610→    if (!doc) return
   611→    const next = !doc.starred
   612→    set({ documents: get().documents.map((d) => (d.id === id ? { ...d, starred: next } : d)) })
   613→    try {
   614→      await fetch(`/api/documents/${id}`, {
   615→        method: "PATCH",
   616→        headers: { "Content-Type": "application/json" },
   617→        body: JSON.stringify({ starred: next }),
   618→      })
   619→    } catch {
   620→      void get().refresh({ silent: true })
   621→    }
   622→  },
   623→
   624→  setTrashed: async (id, trashed) => {
   625→    set({ documents: get().documents.filter((d) => d.id !== id) })
   626→    try {
   627→      await fetch(`/api/documents/${id}`, {
   628→        method: "PATCH",
   629→        headers: { "Content-Type": "application/json" },
   630→        body: JSON.stringify({ trashed }),
   631→      })
   632→      void get().refresh({ silent: true })
   633→    } catch {
   634→      void get().refresh({ silent: true })
   635→    }
   636→  },
   637→
   638→  deleteForever: async (id) => {
   639→    set({ documents: get().documents.filter((d) => d.id !== id) })
   640→    try {
   641→      await fetch(`/api/documents/${id}`, { method: "DELETE" })
   642→    } catch {
   643→      void get().refresh({ silent: true })
   644→    }
   645→  },
   646→
   647→  duplicateDoc: async (id) => {
   648→    try {
   649→      const res = await fetch(`/api/documents/${id}/duplicate`, { method: "POST" })
   650→      if (!res.ok) throw new Error("Duplicate failed")
   651→      await get().refresh({ silent: true })
   652→    } catch {
   653→      void get().refresh({ silent: true })
   654→    }
   655→  },
   656→
   657→  patchDocMeta: (id, patch) => {
   658→    set({ documents: get().documents.map((d) => (d.id === id ? { ...d, ...patch } : d)) })
   659→  },
   660→
   661→  /* ---------------- multi-select + batch ops ---------------- */
   662→
   663→  toggleSelect: (id) => {
   664→    const cur = get().selection
   665→    set({ selection: cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id] })
   666→  },
   667→
   668→  selectAll: (ids) => {
   669→    set({ selection: Array.from(new Set(ids)) })
   670→  },
   671→
   672→  clearSelection: () => set({ selection: [] }),
   673→
   674→  batchOp: async (ids, op, folderId, tagIds) => {
   675→    let ok = 0
   676→    let failed = 0
   677→    for (const id of ids) {
   678→      try {
   679→        let res: Response
   680→        if (op === "tag" && tagIds && tagIds.length > 0) {
   681→          // additive semantics: union each document's existing tags with the
   682→          // selected ones (never removes tags the docs already carry)
   683→          const doc = get().documents.find((d) => d.id === id)
   684→          const existing = (doc?.tags ?? []).map((tg) => tg.id)
   685→          const merged = Array.from(new Set([...existing, ...tagIds]))
   686→          res = await api(`/api/documents/${encodeURIComponent(id)}/tags`, {
   687→            method: "PUT",
   688→            headers: { "Content-Type": "application/json" },
   689→            body: JSON.stringify({ tagIds: merged }),
   690→          })
   691→        } else if (op === "deleteForever") {
   692→          res = await api(`/api/documents/${encodeURIComponent(id)}`, { method: "DELETE" })
   693→        } else {
   694→          const body: Record<string, unknown> =
   695→            op === "star"
   696→              ? { starred: true }
   697→              : op === "unstar"
   698→                ? { starred: false }
   699→                : op === "trash"
   700→                  ? { trashed: true }
   701→                  : op === "restore"
   702→                    ? { trashed: false }
   703→                    : { folderId: folderId ?? null }
   704→          res = await api(`/api/documents/${encodeURIComponent(id)}`, {
   705→            method: "PATCH",
   706→            headers: { "Content-Type": "application/json" },
   707→            body: JSON.stringify(body),
   708→          })
   709→        }
   710→        if (res.ok) ok++
   711→        else failed++
   712→      } catch {
   713→        failed++
   714→      }
   715→    }
   716→    if (ok > 0) {
   717→      await get().refresh({ silent: true })
   718→      if (op === "move") await get().refreshFolders()
   719→      if (op === "tag") await get().loadTags()
   720→    }
   721→    return { ok, failed }
   722→  },
   723→
   724→  /* ---------------- tags ---------------- */
   725→  loadTags: async () => {
   726→    try {
   727→      const res = await fetch("/api/tags")
   728→      if (!res.ok) return
   729→      const data = (await res.json()) as { tags: TagDTO[] }
   730→      set({ tags: data.tags ?? [] })
   731→    } catch {
   732→      // non-fatal
   733→    }
   734→  },
   735→
   736→  setTagFilter: (id) => set({ tagFilter: id }),
   737→
   738→  createTag: async (name, color) => {
   739→    try {
   740→      const res = await fetch("/api/tags", {
   741→        method: "POST",
   742→        headers: { "Content-Type": "application/json" },
   743→        body: JSON.stringify({ name, color }),
   744→      })
   745→      if (!res.ok) return null
   746→      const data = (await res.json()) as { tag: TagDTO }
   747→      await get().loadTags()
   748→      return data.tag
   749→    } catch {
   750→      return null
   751→    }
   752→  },
   753→
   754→  deleteTag: async (id) => {
   755→    const prevDocs = get().documents
   756→    set({
   757→      tags: get().tags.filter((t) => t.id !== id),
   758→      tagFilter: get().tagFilter === id ? null : get().tagFilter,
   759→      documents: prevDocs.map((d) => ({ ...d, tags: (d.tags ?? []).filter((t) => t.id !== id) })),
   760→    })
   761→    try {
   762→      await fetch(`/api/tags/${encodeURIComponent(id)}`, { method: "DELETE" })
   763→    } catch {
   764→      // ignore
   765→    }
   766→    await get().loadTags()
   767→    void get().refresh({ silent: true })
   768→  },
   769→
   770→  setDocTags: async (docId, tagIds) => {
   771→    const prev = get().documents
   772→    const prevDoc = prev.find((d) => d.id === docId)
   773→    if (!prevDoc) return
   774→    const prevIds = (prevDoc.tags ?? []).map((t) => t.id)
   775→    if (prevIds.length === tagIds.length && prevIds.every((id, i) => id === tagIds[i])) return
   776→    const allTags = get().tags
   777→    const optimisticTags = tagIds
   778→      .map((id) => allTags.find((t) => t.id === id))
   779→      .filter((t): t is TagDTO => !!t)
   780→    set({ documents: prev.map((d) => (d.id === docId ? { ...d, tags: optimisticTags } : d)) })
   781→    try {
   782→      const res = await fetch(`/api/documents/${docId}/tags`, {
   783→        method: "PUT",
   784→        headers: { "Content-Type": "application/json" },
   785→        body: JSON.stringify({ tagIds }),
   786→      })
   787→      if (!res.ok) throw new Error("set tags failed")
   788→      const data = (await res.json()) as { tags: TagDTO[] }
   789→      set({
   790→        documents: get().documents.map((d) => (d.id === docId ? { ...d, tags: data.tags ?? [] } : d)),
   791→      })
   792→      await get().loadTags()
   793→    } catch {
   794→      // rollback the optimistic chip change
   795→      set({ documents: prev.map((d) => (d.id === docId ? { ...d, tags: prevDoc.tags ?? [] } : d)) })
   796→    }
   797→  },
   798→
   799→  /* ---------------- folders ---------------- */
   800→  openFolder: (id) => {
   801→    set({ filter: "folder", activeFolderId: id, selection: [] })
   802→    void get().refresh({ silent: true })
   803→  },
   804→
   805→  createFolder: async (name) => {
   806→    try {
   807→      const res = await fetch("/api/folders", {
   808→        method: "POST",
   809→        headers: { "Content-Type": "application/json" },
   810→        body: JSON.stringify({ name }),
   811→      })
   812→      if (!res.ok) return null
   813→      const data = (await res.json()) as { folder: FolderDTO }
   814→      await get().refreshFolders()
   815→      return data.folder
   816→    } catch {
   817→      return null
   818→    }
   819→  },
   820→
   821→  renameFolder: async (id, name) => {
   822→    const optimistic = get().folders.map((f) => (f.id === id ? { ...f, name } : f))
   823→    set({ folders: optimistic })
   824→    try {
   825→      const res = await fetch("/api/folders", {
   826→        method: "PATCH",
   827→        headers: { "Content-Type": "application/json" },
   828→        body: JSON.stringify({ id, name }),
   829→      })
   830→      if (!res.ok) throw new Error("rename failed")
   831→      await get().refreshFolders()
   832→    } catch {
   833→      await get().refreshFolders()
   834→    }
   835→  },
   836→
   837→  deleteFolder: async (id) => {
   838→    set({ folders: get().folders.filter((f) => f.id !== id) })
   839→    try {
   840→      await fetch(`/api/folders?id=${encodeURIComponent(id)}`, { method: "DELETE" })
   841→    } catch {
   842→      // ignore
   843→    }
   844→    // if we were viewing the deleted folder, fall back to all docs
   845→    if (get().activeFolderId === id) {
   846→      set({ filter: "all", activeFolderId: null })
   847→    }
   848→    await get().refresh({ silent: true })
   849→    await get().refreshFolders()
   850→  },
   851→
   852→  moveToFolder: async (docId, folderId) => {
   853→    set({ documents: get().documents.map((d) => (d.id === docId ? { ...d, folderId } : d)) })
   854→    try {
   855→      const res = await fetch(`/api/documents/${docId}`, {
   856→        method: "PATCH",
   857→        headers: { "Content-Type": "application/json" },
   858→        body: JSON.stringify({ folderId }),
   859→      })
   860→      if (!res.ok) throw new Error("move failed")
   861→    } catch {
   862→      void get().refresh({ silent: true })
   863→    }
   864→    await get().refreshFolders()
   865→  },
   866→}))

   868→export function toMeta(doc: DocumentDTO): DocumentMeta {
   869→  return {
   870→    id: doc.id,
   871→    title: doc.title,
   872→    starred: doc.starred,
   873→    trashed: doc.trashed,
   874→    folderId: doc.folderId ?? null,
   875→    tags: doc.tags ?? [],
   876→    snippet: getSnippet(doc.content),
   877→    wordCount: countWords(htmlToText(doc.content)),
   878→    createdAt: doc.createdAt,
   879→    updatedAt: doc.updatedAt,
   880→  }
