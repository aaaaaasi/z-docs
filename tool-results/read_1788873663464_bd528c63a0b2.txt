    61→/* ------------------------------ theme mode hook ----------------------------- */

    63→type ThemeMode = "light" | "dark" | "system"
    64→
    65→function systemPreference(): "light" | "dark" {
    66→  if (typeof window === "undefined" || !window.matchMedia) return "light"
    67→  return window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light"
    68→}
    69→
    70→/**
    71→ * Light / Dark / System selection on top of next-themes' setTheme. The app's
    72→ * ThemeProvider has system resolution disabled, so "System" is resolved here
    73→ * via matchMedia and applied as an explicit light/dark theme — and followed
    74→ * live if the OS preference changes. The chosen mode is mirrored to
    75→ * localStorage and re-synced when the theme is toggled elsewhere (header).
    76→ */
    77→function useThemeMode(): { mode: ThemeMode; setMode: (mode: ThemeMode) => void } {
    78→  const { theme, setTheme } = useTheme()
    79→  const [mode, setModeState] = React.useState<ThemeMode>(() => {
    80→    if (typeof window === "undefined") return "light"
    81→    let own: string | null = null
    82→    try {
    83→      own = window.localStorage.getItem(THEME_MODE_STORAGE_KEY)
    84→    } catch {
    85→      // storage unavailable
    86→    }
    87→    if (own === "light" || own === "dark" || own === "system") {
    88→      const applied = own === "system" ? systemPreference() : own
    89→      if ((theme === "light" || theme === "dark") && theme !== applied) return theme
    90→      return own
    91→    }
    92→    return theme === "dark" ? "dark" : "light"
    93→  })
    94→
    95→  const setMode = React.useCallback(
    96→    (next: ThemeMode) => {
    97→      setModeState(next)
    98→      try {
    99→        window.localStorage.setItem(THEME_MODE_STORAGE_KEY, next)
   100→      } catch {
   101→        // storage unavailable — still apply for this session
   102→      }
   103→      setTheme(next === "system" ? systemPreference() : next)
   104→    },
   105→    [setTheme]
   106→  )
   107→
   108→  // keep the card selection honest when the theme is toggled outside Settings
   109→  React.useEffect(() => {
   110→    if (theme !== "light" && theme !== "dark") return
   111→    const applied = mode === "system" ? systemPreference() : mode
   112→    if (theme !== applied) setModeState(theme)
   113→  }, [theme, mode])
   114→
   115→  // follow OS preference changes while in System mode
   116→  React.useEffect(() => {
   117→    if (mode !== "system") return
   118→    const mq = window.matchMedia("(prefers-color-scheme: dark)")
   119→    const onChange = () => setTheme(mq.matches ? "dark" : "light")
   120→    mq.addEventListener("change", onChange)
   121→    return () => mq.removeEventListener("change", onChange)
   122→  }, [mode, setTheme])
   123→
   124→  return { mode, setMode }
   125→}

   159→function formatBytes(bytes: number): string {
   160→  if (bytes >= 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
   161→  if (bytes >= 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`
   162→  if (bytes >= 1024) return `${(bytes / 1024).toFixed(1)} KB`
   163→  return `${bytes} B`
   164→}

   172→  const [draftName, setDraftName] = React.useState<string | null>(null)
   173→
   174→  const name = draftName ?? user?.name ?? ""
   175→
   176→  // 400ms debounced save of the display name
   177→  React.useEffect(() => {
   178→    if (draftName === null || !user) return
   179→    const timer = setTimeout(() => {
   180→      const trimmed = draftName.trim()
   181→      if (trimmed && trimmed !== user.name) {
   182→        saveLocalUser({ ...user, name: trimmed })
   183→        toast({ title: t("Name saved") })
   184→      }
   185→    }, 400)
   186→    return () => clearTimeout(timer)
   187→  }, [draftName, user, toast, t])
   188→
   189→  if (!user) {
   190→    return (
   191→      <SectionCard title={t("Profile")}>
   192→        <div className="flex items-center gap-4">
   193→          <Skeleton className="h-10 w-10 rounded-full" />

   203→  return (

   452→  React.useEffect(() => {
   453→    try {
   454→      const f = window.localStorage.getItem(FONT_STORAGE_KEY)
   455→      if (f === "sans" || f === "serif") setFont(f)

   459→      // storage unavailable — defaults are fine

   467→      window.localStorage.setItem(FONT_STORAGE_KEY, value)
   468→    } catch {
   469→      // ignore
   470→    }
   471→    toast({ title: value === "sans" ? t("Default font set to Sans") : t("Default font set to Serif") })
   472→  }
   473→
   474→  const saveZoom = (value: string) => {
   475→    if (value !== "100" && value !== "125" && value !== "150") return
   476→    setZoom(value)
   477→    try {
   478→      window.localStorage.setItem(ZOOM_STORAGE_KEY, value)
   479→    } catch {
   480→      // ignore
   481→    }
   482→    toast({ title: t("Default zoom set to {value}%", { value }) })
   483→  }
   484→
   485→  return (
   486→    <SectionCard
   487→      title={t("Workspace defaults")}
   488→      description={t("Preferences used when creating new documents.")}
   489→    >
   490→      <div className="grid gap-4 sm:grid-cols-2">
   491→        <div>
   492→          <Label htmlFor="default-font">{t("Default font")}</Label>
   493→          <Select value={font} onValueChange={saveFont}>
   494→            <SelectTrigger id="default-font" className="mt-1.5 h-10 w-full">
   495→              <SelectValue placeholder={t("Sans")} />
   496→            </SelectTrigger>
   497→            <SelectContent>
   498→              <SelectItem value="sans">{t("Sans (Arial)")}</SelectItem>
   499→              <SelectItem value="serif">{t("Serif (Georgia)")}</SelectItem>
   500→            </SelectContent>
   501→          </Select>
   502→          <p className="mt-1.5 text-xs text-muted-foreground">{t("Used when creating new documents")}</p>
   503→        </div>
   504→        <div>
   505→          <Label htmlFor="default-zoom">{t("Default editor zoom")}</Label>
   506→          <Select value={zoom} onValueChange={saveZoom}>
   507→            <SelectTrigger id="default-zoom" className="mt-1.5 h-10 w-full">
   508→              <SelectValue placeholder="100%" />
   509→            </SelectTrigger>
   510→            <SelectContent>
   511→              <SelectItem value="100">100%</SelectItem>

   517→        </div>
   518→      </div>
   519→    </SectionCard>

   550→  const [counts, setCounts] = React.useState<NonNullable<StoragePayload["counts"]> | null>(null)
   551→  const [bytes, setBytes] = React.useState(0)
   552→  const [quota, setQuota] = React.useState(0)
   553→  const [error, setError] = React.useState<string | null>(null)

   558→      setError(null)
   559→      try {
   560→        // single source of truth — same /api/storage the sidebar reads, so the
   561→        // two surfaces can never disagree (they previously used different
   562→        // quotas: 10 MB here vs 15 GB there)
   563→        const res = await fetch("/api/storage")
   564→        if (!res.ok) throw new Error("Failed to load storage stats")
   565→        const payload = (await res.json()) as StoragePayload
   566→        if (!live) return
   567→        setCounts(payload.counts ?? {})
   568→        setBytes(payload.usedBytes ?? 0)
   569→        setQuota(payload.quotaBytes ?? 0)
   570→      } catch (e) {
   571→        if (!live) return
   572→        setError(e instanceof Error ? e.message : "Something went wrong")
   573→      }
   574→    }
   575→    void load()
   576→    return () => {
   577→      live = false
   578→    }
   579→  }, [])
   580→
   581→  const exportAll = async () => {
   582→    // fetch-based so guest mode is served by the local API shim (an anchor
   583→    // navigation would bypass it and hit the server unauthenticated)
   584→    try {
   585→      const res = await fetch("/api/export")
   586→      if (!res.ok) throw new Error("export failed")
   587→      const blob = await res.blob()
   588→      const url = URL.createObjectURL(blob)
   589→      const a = document.createElement("a")
   590→      a.href = url
   591→      a.download = "z-workspace-export.json"
   592→      document.body.appendChild(a)
   593→      a.click()
   594→      a.remove()
   595→      // async revoke — synchronous revoke can kill the transfer on slow browsers
   596→      setTimeout(() => URL.revokeObjectURL(url), 30_000)
   597→      toast({ title: t("Download started"), description: t("Your workspace is exporting as JSON.") })
   598→    } catch {
   599→      toast({ title: t("Download failed"), description: t("Please try again in a moment.") })
   600→    }
   601→  }
   602→
   603→  const resetLocal = () => {
   604→    for (const key of [
   605→      USER_STORAGE_KEY,
   606→      "zdocs-guest",
   607→      "zdocs-guest-db",
   608→      ACCENT_STORAGE_KEY,
   609→      FONT_STORAGE_KEY,
   610→      ZOOM_STORAGE_KEY,
   611→      THEME_MODE_STORAGE_KEY,
   612→    ]) {
   613→      try {
   614→        window.localStorage.removeItem(key)
   615→      } catch {
   616→        // ignore
   617→      }
   618→    }
   619→    window.location.reload()
   620→  }
   621→
   622→  const usedPercent = counts && quota > 0 ? Math.min(100, (bytes / quota) * 100) : 0
   623→
   624→  return (
   625→    <SectionCard
   626→      title={t("Data & storage")}
   627→      description={t("Everything you create lives in your local Z workspace database.")}
   628→    >
   629→      {error ? (
   630→        <p className="rounded-lg border border-destructive/30 bg-destructive/5 px-4 py-3 text-[13px] text-destructive">
   631→          {t(error)}
   632→        </p>
   633→      ) : counts ? (
   634→        <>
   635→          <div>
   636→            <div className="flex items-baseline justify-between gap-3">
   637→              <p className="text-[13px] font-medium">{t("Workspace storage")}</p>
   638→              <p className="text-xs text-muted-foreground">
   639→                <span className="tnum font-medium text-foreground">{t("{size} used", { size: formatBytes(bytes) })}</span>
   640→                <span className="opacity-70"> · {quota > 0 ? t("{size} quota", { size: formatBytes(quota) }) : ""}</span>
   641→              </p>
   642→            </div>
   643→            <div
   644→              className="mt-2 h-2 overflow-hidden rounded-full bg-muted"
   645→              role="progressbar"
   646→              aria-label={t("Workspace storage used")}
   647→              aria-valuemin={0}
   648→              aria-valuemax={100}
   649→              aria-valuenow={Math.round(usedPercent)}
   650→            >
   651→              <div
   652→                className="h-full rounded-full bg-primary transition-[width] duration-500"
   653→                style={{ width: `${usedPercent}%` }}
   654→              />
   655→            </div>
   656→          </div>
   657→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   658→            {STORAGE_STATS.map((stat) => {
   659→              const Icon = stat.icon
   660→              return (
   661→                <div key={stat.key} className="flex items-center gap-3 rounded-lg border p-3">
   662→                  <Icon className="h-4 w-4 shrink-0 text-muted-foreground" />
   663→                  <div className="min-w-0">
   664→                    <p className="tnum text-[15px] font-medium leading-tight">
   665→                      {counts[stat.key] ?? 0}
   666→                    </p>
   667→                    <p className="text-[11px] text-muted-foreground">{t(stat.label)}</p>
   668→                  </div>
   669→                </div>
   670→              )
   671→            })}
   672→          </div>
   673→        </>
   674→      ) : (
   675→        <>
   676→          <div className="flex items-baseline justify-between">
   677→            <Skeleton className="h-4 w-32" />
   678→            <Skeleton className="h-3.5 w-24" />
   679→          </div>
   680→          <Skeleton className="h-2 w-full rounded-full" />
   681→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   682→            {STORAGE_STATS.map((stat) => (
   683→              <Skeleton key={stat.key} className="h-14 rounded-lg" />
   684→            ))}
   685→          </div>
   686→        </>
   687→      )}
   688→      <div className="flex flex-wrap gap-3">
   689→        <Button onClick={exportAll} className="h-10">
   690→          <Download className="h-4 w-4" />
   691→          {t("Export all your data")}
   692→        </Button>
   693→        <AlertDialog>
   694→          <AlertDialogTrigger asChild>
   695→            <Button
   696→              variant="outline"
   697→              className="h-10 text-destructive hover:bg-destructive/5 hover:text-destructive"
   698→            >
   699→              <RotateCcw className="h-4 w-4" />
   700→              {t("Reset local profile")}
   701→            </Button>
   702→          </AlertDialogTrigger>
   703→          <AlertDialogContent>
   704→            <AlertDialogHeader>
   705→              <AlertDialogTitle>{t("Reset local profile?")}</AlertDialogTitle>
   706→              <AlertDialogDescription>
   707→                {t("This clears your saved name, avatar color, theme, accent and workspace defaults from this browser. Your documents, spreadsheets, decks and forms are kept.")}
   708→              </AlertDialogDescription>
   709→            </AlertDialogHeader>
   710→            <AlertDialogFooter>
   711→              <AlertDialogCancel>{t("Cancel")}</AlertDialogCancel>
   712→              <AlertDialogAction
   713→                onClick={resetLocal}
   714→                className="bg-destructive text-white hover:bg-destructive/90"
   715→              >
   716→                {t("Reset & reload")}
   717→              </AlertDialogAction>
   718→            </AlertDialogFooter>
   719→          </AlertDialogContent>
