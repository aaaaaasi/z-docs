    58→    58→/* ------------------------------ theme mode hook ----------------------------- */
    59→    59→
    60→    60→type ThemeMode = "light" | "dark" | "system"
    61→    61→
    62→    62→function systemPreference(): "light" | "dark" {
    63→    63→  if (typeof window === "undefined" || !window.matchMedia) return "light"
    64→    64→  return window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light"
    65→    65→}
    66→    66→
    67→    67→/**
    68→    68→ * Light / Dark / System selection on top of next-themes' setTheme. The app's
    69→    69→ * ThemeProvider has system resolution disabled, so "System" is resolved here
    70→    70→ * via matchMedia and applied as an explicit light/dark theme — and followed
    71→    71→ * live if the OS preference changes. The chosen mode is mirrored to
    72→    72→ * localStorage and re-synced when the theme is toggled elsewhere (header).
    73→    73→ */
    74→    74→function useThemeMode(): { mode: ThemeMode; setMode: (mode: ThemeMode) => void } {
    75→    75→  const { theme, setTheme } = useTheme()
    76→    76→  const [mode, setModeState] = React.useState<ThemeMode>(() => {
    77→    77→    if (typeof window === "undefined") return "light"
    78→    78→    let own: string | null = null
    79→    79→    try {
    80→    80→      own = window.localStorage.getItem(THEME_MODE_STORAGE_KEY)
    81→    81→    } catch {
    82→    82→      // storage unavailable
    83→    83→    }
    84→    84→    if (own === "light" || own === "dark" || own === "system") {
    85→    85→      const applied = own === "system" ? systemPreference() : own
    86→    86→      if ((theme === "light" || theme === "dark") && theme !== applied) return theme
    87→    87→      return own
    88→    88→    }
    89→    89→    return theme === "dark" ? "dark" : "light"
    90→    90→  })
    91→    91→
    92→    92→  const setMode = React.useCallback(
    93→    93→    (next: ThemeMode) => {
    94→    94→      setModeState(next)
    95→    95→      try {
    96→    96→        window.localStorage.setItem(THEME_MODE_STORAGE_KEY, next)
    97→    97→      } catch {
    98→    98→        // storage unavailable — still apply for this session
    99→    99→      }
   100→   100→      setTheme(next === "system" ? systemPreference() : next)
   101→   101→    },
   102→   102→    [setTheme]
   103→   103→  )
   104→   104→
   105→   105→  // keep the card selection honest when the theme is toggled outside Settings
   106→   106→  React.useEffect(() => {
   107→   107→    if (theme !== "light" && theme !== "dark") return
   108→   108→    const applied = mode === "system" ? systemPreference() : mode
   109→   109→    if (theme !== applied) setModeState(theme)
   110→   110→  }, [theme, mode])
   111→   111→
   112→   112→  // follow OS preference changes while in System mode
   113→   113→  React.useEffect(() => {
   114→   114→    if (mode !== "system") return
   115→   115→    const mq = window.matchMedia("(prefers-color-scheme: dark)")
   116→   116→    const onChange = () => setTheme(mq.matches ? "dark" : "light")
   117→   117→    mq.addEventListener("change", onChange)
   118→   118→    return () => mq.removeEventListener("change", onChange)
   119→   119→  }, [mode, setTheme])
   120→   120→
   121→   121→  return { mode, setMode }
   122→   122→}

   400→   400→      const f = window.localStorage.getItem(FONT_STORAGE_KEY)

   405→   405→      // storage unavailable — defaults are fine

   407→   407→  }, [])

   413→   413→      window.localStorage.setItem(FONT_STORAGE_KEY, value)

   415→   415→      // ignore
   416→   416→    }
   417→   417→    toast({ title: value === "sans" ? "Default font set to Sans" : "Default font set to Serif" })
   418→   418→  }
   419→   419→
   420→   420→  const saveZoom = (value: string) => {
   421→   421→    if (value !== "100" && value !== "125" && value !== "150") return
   422→   422→    setZoom(value)
   423→   423→    try {
   424→   424→      window.localStorage.setItem(ZOOM_STORAGE_KEY, value)
   425→   425→    } catch {
   426→   426→      // ignore
   427→   427→    }
   428→   428→    toast({ title: `Default zoom set to ${value}%` })
   429→   429→  }
   430→   430→
   431→   431→  return (
   432→   432→    <SectionCard
   433→   433→      title="Workspace defaults"
   434→   434→      description="Preferences used when creating new documents."
   435→   435→    >
   436→   436→      <div className="grid gap-4 sm:grid-cols-2">
   437→   437→        <div>
   438→   438→          <Label htmlFor="default-font">Default font</Label>
   439→   439→          <Select value={font} onValueChange={saveFont}>
   440→   440→            <SelectTrigger id="default-font" className="mt-1.5 h-10 w-full">
   441→   441→              <SelectValue placeholder="Sans" />
   442→   442→            </SelectTrigger>
   443→   443→            <SelectContent>
   444→   444→              <SelectItem value="sans">Sans (Arial)</SelectItem>
   445→   445→              <SelectItem value="serif">Serif (Georgia)</SelectItem>
   446→   446→            </SelectContent>
   447→   447→          </Select>
   448→   448→          <p className="mt-1.5 text-xs text-muted-foreground">Used when creating new documents</p>
   449→   449→        </div>
   450→   450→        <div>
   451→   451→          <Label htmlFor="default-zoom">Default editor zoom</Label>
   452→   452→          <Select value={zoom} onValueChange={saveZoom}>
   453→   453→            <SelectTrigger id="default-zoom" className="mt-1.5 h-10 w-full">
   454→   454→              <SelectValue placeholder="100%" />
   455→   455→            </SelectTrigger>
   456→   456→            <SelectContent>
   457→   457→              <SelectItem value="100">100%</SelectItem>
   458→   458→              <SelectItem value="125">125%</SelectItem>
   459→   459→              <SelectItem value="150">150%</SelectItem>
   460→   460→            </SelectContent>
   461→   461→          </Select>
   462→   462→          <p className="mt-1.5 text-xs text-muted-foreground">Used when creating new documents</p>
   463→   463→        </div>

   499→   499→  React.useEffect(() => {

   502→   502→      setError(null)
   503→   503→      try {
   504→   504→        const res = await fetch("/api/export")

   589→   589→                  <Icon className="h-4 w-4 shrink-0 text-muted-foreground" />
   590→   590→                  <div className="min-w-0">
   591→   591→                    <p className="tnum text-[15px] font-medium leading-tight">
   592→   592→                      {counts[stat.key] ?? 0}
