   168→   168→const TERMINATORS = new Set(["\u3002", "\uff01", "\uff1f", "!", "?", "\u2026"])

   179→   179→/**
   180→   180→ * Does `buf` (which already ends with the candidate char) end a sentence?
   181→   181→ * next1/next2 are the two chars that follow in the source text.
   182→   182→ */
   183→   183→function endsSentence(buf: string, next1: string, next2: string): boolean {
   184→   184→  const ch = buf[buf.length - 1]
   185→   185→  if (ch === undefined) return false
   186→   186→  if (TERMINATORS.has(ch)) return true
   187→   187→  if (ch !== ".") return false
   188→   188→  const prev = buf[buf.length - 2] ?? ""
   189→   189→  if (/[0-9]/.test(prev) && /[0-9]/.test(next1)) return false // 3.14
   190→   190→  const m = /([A-Za-z][A-Za-z.]*)$/.exec(buf)
   191→   191→  if (m) {
   192→   192→    const candidate = m[1].replace(/\.+$/, "").toLowerCase()
   193→   193→    if (ABBREVIATIONS.has(candidate)) return false
   194→   194→    // first dot of "e.g." / "i.e."
   195→   195→    if (candidate === "e" && next1 === "g" && next2 === ".") return false
   196→   196→    if (candidate === "i" && next1 === "e" && next2 === ".") return false
   197→   197→  }
   198→   198→  return true
   199→   199→}

   201→   201→function isPlainPeriod(text: string, j: number): boolean {
   202→   202→  const prev = text[j - 1] ?? ""
   203→   203→  const next = text[j + 1] ?? ""
   204→   204→  return !(prev >= "0" && prev <= "9" && next >= "0" && next <= "9")
   205→   205→}

   207→   207→function splitSentences(text: string): string[] {
   208→   208→  const out: string[] = []
   209→   209→  let buf = ""
   210→   210→  const push = () => {
   211→   211→    const s = buf.trim()
   212→   212→    buf = ""
   213→   213→    if (s && SENTENCE_WORD_RE.test(s)) out.push(s)
   214→   214→  }
   215→   215→  const n = text.length
   216→   216→  let i = 0
   217→   217→  while (i < n) {
   218→   218→    const ch = text[i]
   219→   219→    if (ch === "\n" || ch === "\r") {
   220→   220→      push()
   221→   221→      i++
   222→   222→      continue
   223→   223→    }
   224→   224→    buf += ch
   225→   225→    if (endsSentence(buf, text[i + 1] ?? "", text[i + 2] ?? "")) {
   226→   226→      // swallow terminator runs ("?!", "...", "。……") and glued closers (”」』）
   227→   227→      let j = i + 1
   228→   228→      while (j < n) {
   229→   229→        const c = text[j]
   230→   230→        if (c === "\n" || c === "\r") break
   231→   231→        const isTerm = TERMINATORS.has(c) || (c === "." && isPlainPeriod(text, j))
   232→   232→        if (!isTerm && !TRAILING_CLOSERS.has(c)) break
   233→   233→        buf += c
   234→   234→        j++
   235→   235→      }
   236→   236→      push()
   237→   237→      i = j
   238→   238→      continue
   239→   239→    }
   240→   240→    i++
   241→   241→  }
   242→   242→  push()
   243→   243→  return out
   244→   244→}

   299→   299→function countSyllables(word: string): number {
   300→   300→  const w = word.toLowerCase()
   301→   301→  if (!/^[a-z]+$/.test(w)) return 1
   302→   302→  if (w.length <= 3) return 1
   303→   303→  // silent trailing e (table/sea keep theirs)
   304→   304→  const base = /e$/.test(w) && !/(le|ee|ye)$/.test(w) ? w.slice(0, -1) : w
   305→   305→  const groups = base.match(/[aeiouy]+/g)
   306→   306→  return Math.max(1, groups ? groups.length : 1)
   307→   307→}

   309→   309→function computeTextComplexity(tokens: string[], sentences: string[]): TextComplexity {
   310→   310→  if (tokens.length === 0) return { grade: 0, label: LABEL_SUITABLE_GRADE, isCJK: true }
   311→   311→  const latinTokens = tokens.filter(isLatinToken)
   312→   312→  const isCJK = latinTokens.length / tokens.length <= 0.5
   313→   313→  if (isCJK) {
   314→   314→    // weighted estimate: avg sentence length (chars count as words) + rare-char rate
   315→   315→    const avgLen = sentences.length > 0 ? tokens.length / sentences.length : tokens.length
   316→   316→    const grade = avgLen * 1.0 + zhRareRate(tokens) * 15
   317→   317→    return { grade: r1(clamp(grade, 1, 16)), label: LABEL_SUITABLE_GRADE, isCJK: true }
   318→   318→  }
   319→   319→  // Flesch–Kincaid grade level
   320→   320→  const wps = sentences.length > 0 ? latinTokens.length / sentences.length : latinTokens.length
   321→   321→  const syllables = latinTokens.reduce((s, w) => s + countSyllables(w), 0)
   322→   322→  const spw = latinTokens.length > 0 ? syllables / latinTokens.length : 1
   323→   323→  const grade = 0.39 * wps + 11.8 * spw - 15.59
   324→   324→  return { grade: r1(clamp(grade, 1, 20)), label: LABEL_SUITABLE_GRADE, isCJK: false }
   325→   325→}

   327→   327→// ---------------------------------------------------------------------------
   328→   328→// 2. Vocabulary diversity
   329→   329→// ---------------------------------------------------------------------------

   331→   331→function computeVocabDiversity(tokens: string[]): number {
   332→   332→  if (tokens.length === 0) return 0
   333→   333→  const uniq = new Set(tokens.map((t) => t.toLowerCase()))
   334→   334→  return r1((uniq.size / tokens.length) * 100)
   335→   335→}

   341→   341→function computeSentenceLengths(sentences: string[]): SentenceLengths {
   342→   342→  const counts = { short: 0, medium: 0, long: 0, veryLong: 0 }
   343→   343→  for (const s of sentences) {
   344→   344→    const wc = tokenize(s).length
   345→   345→    if (wc <= 5) counts.short++
   346→   346→    else if (wc <= 14) counts.medium++
   347→   347→    else if (wc <= 25) counts.long++
   348→   348→    else counts.veryLong++
   349→   349→  }
   350→   350→  const total = sentences.length
   351→   351→  const pct = (c: number): number => (total > 0 ? r1((c / total) * 100) : 0)
   352→   352→  return {
   353→   353→    short: { count: counts.short, pct: pct(counts.short) },
   354→   354→    medium: { count: counts.medium, pct: pct(counts.medium) },
   355→   355→    long: { count: counts.long, pct: pct(counts.long) },
   356→   356→    veryLong: { count: counts.veryLong, pct: pct(counts.veryLong) },
   357→   357→  }
   358→   358→}

   361→   361→// 5. Paragraph density
   362→   362→// ---------------------------------------------------------------------------
   363→   363→
   364→   364→function computeParagraphDensity(text: string): ParagraphDensityItem[] {
   365→   365→  const out: ParagraphDensityItem[] = []
   366→   366→  for (const para of splitParagraphs(text)) {
   367→   367→    const sentences = splitSentences(para)
   368→   368→    const tokens = tokenize(para)
   369→   369→    if (sentences.length === 0 || tokens.length === 0) continue
   370→   370→    const avgSentenceLen = tokens.length / sentences.length
   371→   371→    const latinTokens = tokens.filter(isLatinToken)
   372→   372→    const latinRatio = latinTokens.length / tokens.length
   373→   373→    let weight: number
   374→   374→    if (latinRatio > 0.5) {
   375→   375→      // long latin words raise the load
   376→   376→      const avgWordLen =
   377→   377→        latinTokens.reduce((s, t) => s + t.length, 0) / Math.max(1, latinTokens.length)
   378→   378→      weight = clamp(avgWordLen / 4.7, 0.6, 1.8)
   379→   379→    } else {
   380→   380→      // rescale CJK chars (≈1.8 chars/word) + rare-char complexity
   381→   381→      weight = 0.55 * (1 + zhRareRate(tokens) * 2)
   382→   382→    }
   383→   383→    const load = avgSentenceLen * weight
   384→   384→    const score: DensityScore =
   385→   385→      load < 7 ? "simple" : load < 13 ? "medium" : load < 20 ? "complex" : "dense"
   386→   386→    out.push({ text: para, score })
   387→   387→  }
   388→   388→  return out
   389→   389→}

   468→   468→function computePassiveVoice(sentences: string[]): PassiveVoiceResult {

   484→   484→    }

   486→   486→  }

   513→   513→/** 地-initial nouns (地图、地方…) — 地 preceded by a CJK char and followed by these is not an adverb */
   514→   514→const ZH_AFTER_DI_BLOCK = new Set(
   515→   515→  "图方球址区铁上下面板步位带壳狱道雷窖毯契租瓜砖摊基界矿产形势理利籍名点类样色貌".split("")

   520→   520→  const out: string[] = []
   521→   521→  let m: RegExpExecArray | null
   522→   522→  ZH_DE_RE.lastIndex = 0
   523→   523→  while ((m = ZH_DE_RE.exec(text)) !== null) {
   524→   524→    const phrase = m[1]
   525→   525→    const stem = phrase.slice(0, -1)
   526→   526→    if (ZH_DE_BLOCKLIST.some((w) => phrase.endsWith(w))) continue
   527→   527→    const after = text[m.index + m[0].length] ?? ""
   528→   528→    if (ZH_AFTER_DI_BLOCK.has(after)) continue
   529→   529→    let blocked = false
   530→   530→    for (const c of ZH_DE_PRECEDING_BLOCK) {
   531→   531→      if (stem.endsWith(c)) {
   532→   532→        blocked = true
   533→   533→        break
   534→   534→      }
   535→   535→    }
   536→   536→    if (blocked) continue
   537→   537→    out.push(phrase)
   538→   538→  }
   539→   539→  return out
   540→   540→}

   559→   559→}

   593→   593→function cjkRunsOf(sentence: string): string[] {
   594→   594→  return sentence.match(/[\u4e00-\u9fff]+/g) ?? []
   595→   595→}
   596→   596→
   597→   597→/**
   598→   598→ * Maximal shared CJK substrings (≥2 chars, not all stopwords) between two
   599→   599→ * sentences — the zh counterpart of a shared content word.
   600→   600→ */
   601→   601→function sharedCjkRuns(a: string, b: string): string[] {
   602→   602→  const bigramsB = new Set<string>()
   603→   603→  for (const run of cjkRunsOf(b)) {
   604→   604→    for (let i = 0; i + 1 < run.length; i++) bigramsB.add(run.slice(i, i + 2))
   605→   605→  }
   606→   606→  if (bigramsB.size === 0) return []
   607→   607→  const raw: string[] = []
   608→   608→  for (const run of cjkRunsOf(a)) {
   609→   609→    let i = 0
   610→   610→    while (i + 1 < run.length) {
   611→   611→      if (!bigramsB.has(run.slice(i, i + 2))) {
   612→   612→        i++
   613→   613→        continue
   614→   614→      }
   615→   615→      // extend while the bigram ending at j+1 is also shared
   616→   616→      let j = i + 2
   617→   617→      while (j < run.length && bigramsB.has(run.slice(j - 1, j + 1))) j++
   618→   618→      const seg = run.slice(i, j)
   619→   619→      const hasContent = [...seg].some((c) => !ZH_STOPWORDS.has(c))
   620→   620→      if (hasContent) raw.push(seg)
   621→   621→      i = j
   622→   622→    }
   623→   623→  }
   624→   624→  // keep only maximal runs (longest first, drop contained ones)
   625→   625→  raw.sort((x, y) => y.length - x.length)
   626→   626→  const merged: string[] = []
   627→   627→  for (const seg of raw) {
   628→   628→    if (merged.some((r) => r.includes(seg))) continue
   629→   629→    merged.push(seg)
   630→   630→  }
   631→   631→  return merged
   632→   632→}

   634→   634→function computeWordEcho(sentences: string[]): WordEchoResult {
   635→   635→  let count = 0
   636→   636→  const pairs: EchoPair[] = []
   637→   637→  for (let i = 0; i + 1 < sentences.length; i++) {
   638→   638→    const a = sentences[i]
   639→   639→    const b = sentences[i + 1]
   640→   640→    const setA = latinContentSet(a)
   641→   641→    const setB = latinContentSet(b)
   642→   642→    const sharedLatin: string[] = []
   643→   643→    for (const w of setA) if (setB.has(w)) sharedLatin.push(w)
   644→   644→    const sharedCjk = sharedCjkRuns(a, b)
   645→   645→    if (sharedLatin.length + sharedCjk.length === 0) continue
   646→   646→    count++
   647→   647→    if (pairs.length < 12) {
   648→   648→      pairs.push({ a, b, shared: [...sharedLatin.slice(0, 3), ...sharedCjk.slice(0, 3)] })
   649→   649→    }
   650→   650→  }
   651→   651→  return { count, pairs }
   652→   652→}

   658→   658→function joinTokens(tokens: string[]): string {
   659→   659→  let out = ""
   660→   660→  let prevLatin = false
   661→   661→  for (const tk of tokens) {
   662→   662→    const latin = isLatinToken(tk)
   663→   663→    if (out && (latin || prevLatin)) out += " "
   664→   664→    out += tk
   665→   665→    prevLatin = latin
   666→   666→  }
   667→   667→  return out
   668→   668→}

   670→   670→function containsTokens(haystack: string[], needle: string[]): boolean {
   671→   671→  if (needle.length === 0 || needle.length > haystack.length) return false
   672→   672→  outer: for (let i = 0; i + needle.length <= haystack.length; i++) {
   673→   673→    for (let j = 0; j < needle.length; j++) {
   674→   674→      if (haystack[i + j] !== needle[j]) continue outer
   675→   675→    }
   676→   676→    return true
   677→   677→  }
   678→   678→  return false
   679→   679→}
   680→   680→
   681→   681→const PHRASE_CHUNK_SPLIT_RE = /[,,\uff1b;\uff1a:()\uff08\uff09[\]\u3010\u3011\u2014\u2013]/
   682→   682→
   683→   683→function computeRepeatedPhrases(sentences: string[]): RepeatedPhrase[] {
   684→   684→  const counts = new Map<string, number>()
   685→   685→  const displays = new Map<string, string>()
   686→   686→  for (const sentence of sentences) {
   687→   687→    // phrases should not straddle commas/parens/dashes
   688→   688→    for (const chunk of sentence.split(PHRASE_CHUNK_SPLIT_RE)) {
   689→   689→      const tokens = tokenize(chunk).map((t) => (isLatinToken(t) ? t.toLowerCase() : t))
   690→   690→      for (let n = 2; n <= 6; n++) {
   691→   691→        for (let i = 0; i + n <= tokens.length; i++) {
   692→   692→          const slice = tokens.slice(i, i + n)
   693→   693→          if (slice.every(isStopword)) continue
   694→   694→          const key = slice.join("\u0001")
   695→   695→          counts.set(key, (counts.get(key) ?? 0) + 1)
   696→   696→          if (!displays.has(key)) displays.set(key, joinTokens(slice))
   697→   697→        }
   698→   698→      }
   699→   699→    }
   700→   700→  }
   701→   701→  const entries = [...counts.entries()].filter(([, c]) => c >= 3)
   702→   702→  entries.sort((x, y) => {
   703→   703→    const lenX = x[0].split("\u0001").length
   704→   704→    const lenY = y[0].split("\u0001").length
   705→   705→    return y[1] - x[1] || lenY - lenX || (x[0] < y[0] ? -1 : 1)
   706→   706→  })
   707→   707→  const out: RepeatedPhrase[] = []
   708→   708→  for (const [key, c] of entries) {
   709→   709→    if (out.length >= 10) break
   710→   710→    const phrase = displays.get(key) ?? key
   711→   711→    const phraseTokens = key.split("\u0001")
   712→   712→    const subsumed = out.some(
   713→   713→      (kept) => kept.count >= c && containsTokens(tokenize(kept.phrase), phraseTokens)
   714→   714→    )
   715→   715→    if (subsumed) continue
   716→   716→    out.push({ phrase, count: c })
   717→   717→  }
   718→   718→  return out
   719→   719→}
   720→   720→
   721→   721→// ---------------------------------------------------------------------------
   722→   722→// 12. Dialogue balance
   723→   723→// ---------------------------------------------------------------------------
   724→   724→
   725→   725→const QUOTE_OPENER_TO_CLOSER: Record<string, string> = {
   726→   726→  "\u201c": "\u201d", // “ ”
   727→   727→  "\u2018": "\u2019", // ‘ ’
   728→   728→  "\u300c": "\u300d", // 「 」
   729→   729→  "\u300e": "\u300f", // 『 』
   730→   730→}
   731→   731→const CLOSER_TO_OPENER: Record<string, string> = {
   732→   732→  "\u201d": "\u201c",
   733→   733→  "\u2019": "\u2018",
   734→   734→  "\u300d": "\u300c",
   735→   735→  "\u300f": "\u300e",
   736→   736→}
   737→   737→const QUOTE_ANY_RE = /[\u201c\u201d\u2018\u2019\u300c\u300d\u300e\u300f"']/
   738→   738→
   739→   739→function countNonSpace(s: string): number {
   740→   740→  let n = 0
   741→   741→  for (const ch of s) if (ch !== " " && ch !== "\t" && ch !== "\n" && ch !== "\r") n++
   742→   742→  return n
   743→   743→}
   744→   744→
   745→   745→function scanQuotes(text: string): { innerNonSpace: number; pairCount: number } {
   746→   746→  let innerNonSpace = 0
   747→   747→  let pairCount = 0
   748→   748→  const stacks = new Map<string, number[]>()
   749→   749→  for (let i = 0; i < text.length; i++) {
   750→   750→    const ch = text[i]
   751→   751→    if (ch === "'") {
   752→   752→      // apostrophe inside a contraction — never a quote
   753→   753→      const prev = text[i - 1] ?? ""
   754→   754→      const next = text[i + 1] ?? ""
   755→   755→      if (/[A-Za-z]/.test(prev) && /[A-Za-z]/.test(next)) continue
   756→   756→    }
   757→   757→    const closer = QUOTE_OPENER_TO_CLOSER[ch]
   758→   758→    if (closer !== undefined) {
   759→   759→      const stack = stacks.get(ch) ?? []
   760→   760→      stack.push(i)
   761→   761→      stacks.set(ch, stack)
   762→   762→      continue
   763→   763→    }
   764→   764→    const opener = CLOSER_TO_OPENER[ch]
   765→   765→    if (opener !== undefined) {
   766→   766→      const stack = stacks.get(opener)
   767→   767→      if (stack && stack.length > 0) {
   768→   768→        const start = stack.pop() as number
   769→   769→        innerNonSpace += countNonSpace(text.slice(start + 1, i))
   770→   770→        pairCount++
   771→   771→      }
   772→   772→      continue
   773→   773→    }
   774→   774→    if (ch === '"' || ch === "'") {
   775→   775→      const stack = stacks.get(ch)
   776→   776→      if (stack && stack.length > 0) {
   777→   777→        const start = stack.pop() as number
   778→   778→        innerNonSpace += countNonSpace(text.slice(start + 1, i))
   779→   779→        pairCount++
   780→   780→      } else {
   781→   781→        stacks.set(ch, [i])
   782→   782→      }
   783→   783→    }
   784→   784→  }
   785→   785→  return { innerNonSpace, pairCount }
   786→   786→}
   787→   787→
   788→   788→function computeDialogueBalance(
   789→   789→  text: string,
   790→   790→  sentences: string[]
   791→   791→): DialogueBalanceResult {
   792→   792→  const { innerNonSpace } = scanQuotes(text)
   793→   793→  const total = countNonSpace(text)
   794→   794→  const dialoguePct = total > 0 ? r1((innerNonSpace / total) * 100) : 0
   795→   795→  const narrationPct = Math.max(0, r1(100 - dialoguePct))
   796→   796→  const dialogueSentences = sentences
   797→   797→    .filter((s) => QUOTE_ANY_RE.test(s))
   798→   798→    .slice(0, 60)
   799→   799→  return { dialoguePct, narrationPct, dialogueSentences }
   800→   800→}

   802→   802→// ---------------------------------------------------------------------------
   803→   803→// 13. Point of view
   804→   804→// ---------------------------------------------------------------------------
   805→   805→
   806→   806→const POV_FIRST_EN = /\b(i|we|me|us|my|our|mine|ours|myself|ourselves)\b/i
   807→   807→const POV_SECOND_EN = /\b(you|your|yours)\b/i
   808→   808→const POV_THIRD_EN = /\b(he|she|it|they|him|her|them|his|hers|their|theirs|its)\b/i
   809→   809→const POV_FIRST_ZH = ["我们", "咱们", "我"]
   810→   810→const POV_SECOND_ZH = ["你们", "您", "你"]
   811→   811→const POV_THIRD_ZH = ["他们", "她们", "它们", "他", "她", "它"]
   812→   812→
   813→   813→type PovKind = "first" | "second" | "third"
   814→   814→
   815→   815→function povCategory(sentence: string): PovKind | null {
   816→   816→  let best: PovKind | null = null
   817→   817→  let bestIdx = Infinity
   818→   818→  const latin: Array<[PovKind, RegExp]> = [
   819→   819→    ["first", POV_FIRST_EN],
   820→   820→    ["second", POV_SECOND_EN],
   821→   821→    ["third", POV_THIRD_EN],
   822→   822→  ]
   823→   823→  for (const [kind, re] of latin) {
   824→   824→    const m = re.exec(sentence)
   825→   825→    if (m && m.index < bestIdx) {
   826→   826→      bestIdx = m.index
   827→   827→      best = kind
   828→   828→    }
   829→   829→  }
   830→   830→  const zh: Array<[PovKind, string[]]> = [
   831→   831→    ["first", POV_FIRST_ZH],
   832→   832→    ["second", POV_SECOND_ZH],
   833→   833→    ["third", POV_THIRD_ZH],
   834→   834→  ]
   835→   835→  for (const [kind, markers] of zh) {
   836→   836→    for (const mk of markers) {
   837→   837→      const idx = sentence.indexOf(mk)
   838→   838→      if (idx !== -1 && idx < bestIdx) {
   839→   839→        bestIdx = idx
   840→   840→        best = kind
   841→   841→      }
   842→   842→    }
   843→   843→  }
   844→   844→  return best
   845→   845→}
   846→   846→
   847→   847→function computePointOfView(sentences: string[]): PointOfViewResult {
   848→   848→  const result: PointOfViewResult = {
   849→   849→    first: 0,
   850→   850→    second: 0,
   851→   851→    third: 0,
   852→   852→    total: sentences.length,
   853→   853→    examples: { first: [], second: [], third: [] },
   854→   854→  }
   855→   855→  for (const s of sentences) {
   856→   856→    const kind = povCategory(s)
   857→   857→    if (!kind) continue
   858→   858→    result[kind]++
   859→   859→    const list = result.examples[kind]
   860→   860→    if (list.length < 5) list.push(s)
   861→   861→  }
   862→   862→  return result
   863→   863→}

   869→   869→function countMatches(text: string, re: RegExp): number {
   870→   870→  const g = new RegExp(re.source, re.flags.includes("g") ? re.flags : re.flags + "g")
   871→   871→  return (text.match(g) ?? []).length
   872→   872→}
   873→   873→
   874→   874→function computePunctuationHabits(
   875→   875→  text: string,
   876→   876→  totalWords: number,
   877→   877→  quotePairs: number
   878→   878→): PunctuationHabit[] {
   879→   879→  const groups: Array<{ char: string; label: string; count: number }> = [
   880→   880→    { char: "\uff01", label: "Exclamation marks", count: countMatches(text, /[\uff01!]/g) },
   881→   881→    { char: "\uff1f", label: "Question marks", count: countMatches(text, /[\uff1f?]/g) },
   882→   882→    { char: "\uff1b", label: "Semicolons", count: countMatches(text, /[\uff1b;]/g) },
   883→   883→    { char: "\uff1a", label: "Colons", count: countMatches(text, /[\uff1a:]/g) },
   884→   884→    { char: "\u2014", label: "Dashes", count: countMatches(text, /[\u2014\u2013]/g) },
   885→   885→    { char: "\u3001", label: "Enumeration commas", count: countMatches(text, /\u3001/g) },
   886→   886→    { char: "\u3002", label: "Periods", count: countMatches(text, /\u3002/g) },
   887→   887→    { char: "\u2026", label: "Ellipses", count: countMatches(text, /(\u2026{1,2}|\.{3,6})/g) },
   888→   888→    { char: "\u201c\u201d", label: "Quotation marks", count: quotePairs },
   889→   889→    { char: "()", label: "Parentheses", count: countMatches(text, /[\uff08(]/g) },
   890→   890→  ]
   891→   891→  const per = (c: number): number => (totalWords > 0 ? r1((c / totalWords) * 1000) : 0)
   892→   892→  return groups
   893→   893→    .filter((g) => g.count > 0)
   894→   894→    .map((g) => ({ char: g.char, label: g.label, per1000: per(g.count) }))
   895→   895→    .sort((a, b) => b.per1000 - a.per1000)
   896→   896→    .slice(0, 8)
   897→   897→}

   903→   903→export function computeInsights(text: string): InsightsResult {
   904→   904→  const sentences = splitSentences(text)
   905→   905→  const tokens = tokenize(text)
   906→   906→  const totalWords = tokens.length
   907→   907→  const paragraphs = splitParagraphs(text)
   908→   908→
   909→   909→  const summary: SummaryResult = {
   910→   910→    sentences: sentences.length,
   911→   911→    words: totalWords,
   912→   912→    paragraphs: paragraphs.length,
   913→   913→    avgSentenceLen:
   914→   914→      sentences.length > 0 ? r1(totalWords / sentences.length) : 0,
   915→   915→  }
   916→   916→
   917→   917→  const quoteScan = scanQuotes(text)
   918→   918→
   919→   919→  return {
   920→   920→    summary,
   921→   921→    textComplexity: computeTextComplexity(tokens, sentences),
   922→   922→    vocabDiversity: computeVocabDiversity(tokens),
   923→   923→    sentenceLengths: computeSentenceLengths(sentences),
   924→   924→    sentenceRhythm: sentences.map((s) => ({ sentence: s, wordCount: tokenize(s).length })),
   925→   925→    paragraphDensity: computeParagraphDensity(text),
   926→   926→    sentenceOpeners: computeSentenceOpeners(sentences),
   927→   927→    passiveVoice: computePassiveVoice(sentences),
   928→   928→    adverbs: computeAdverbs(text, tokens),
   929→   929→    wordFrequency: computeWordFrequency(tokens),
   930→   930→    wordEcho: computeWordEcho(sentences),
   931→   931→    repeatedPhrases: computeRepeatedPhrases(sentences),

   933→   933→    pointOfView: computePointOfView(sentences),
   934→   934→    punctuationHabits: computePunctuationHabits(text, totalWords, quoteScan.pairCount),
   935→   935→  }
   936→   936→}
