    44→function Composer({
    45→  quote,
    46→  submitting,
    47→  onSubmit,
    48→  onCancel,
    49→}: {
    50→  quote: string | null
    51→  submitting: boolean
    52→  onSubmit: (text: string) => void
    53→  onCancel: () => void
    54→}) {
    55→  const [text, setText] = React.useState("")
    56→  const ref = React.useRef<HTMLTextAreaElement | null>(null)
    57→
    58→  React.useEffect(() => {
    59→    setText("")
    60→    if (quote != null) {
    61→      // focus the textarea when a new quote is captured
    62→      const t = setTimeout(() => ref.current?.focus(), 60)
    63→      return () => clearTimeout(t)
    64→    }
    65→  }, [quote])
    66→
    67→  if (quote == null) return null
    68→
    69→  return (
    70→    <div className="animate-fade-in rounded-lg border bg-card p-3" data-testid="comment-composer">
    71→      <div className="mb-2 flex items-start gap-2 rounded-lg bg-muted/70 p-2">
    72→        <Quote className="mt-0.5 h-3.5 w-3.5 shrink-0 text-muted-foreground" />
    73→        <p className="line-clamp-3 text-xs italic leading-snug text-muted-foreground">{quote}</p>
    74→      </div>
    75→      <Textarea
    76→        ref={ref}
    77→        value={text}
    78→        onChange={(e) => setText(e.target.value)}
    79→        onKeyDown={(e) => {
    80→          if (e.key === "Enter" && (e.ctrlKey || e.metaKey) && text.trim()) {
    81→            e.preventDefault()
    82→            onSubmit(text.trim())
    83→          }
    84→          if (e.key === "Escape") onCancel()
    85→        }}
    86→        placeholder="Write a comment… (Ctrl+Enter to post)"
    87→        className="min-h-[72px] max-h-56 resize-none overflow-y-auto border-none bg-transparent p-0 text-sm shadow-none focus-visible:ring-0"
    88→        maxLength={2000}
    89→        aria-label="Comment text"
    90→      />
    91→      <div className="mt-2 flex items-center justify-end gap-2">
    92→        <Button variant="ghost" size="sm" className="h-8 rounded-md text-muted-foreground" onClick={onCancel}>
    93→          Cancel
    94→        </Button>
    95→        <Button
    96→          size="sm"
    97→          className="h-8 gap-1.5 rounded-md px-4"
    98→          disabled={!text.trim() || submitting}
    99→          onClick={() => onSubmit(text.trim())}
   100→        >
   101→          {submitting ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <MessageSquarePlus className="h-3.5 w-3.5" />}
   102→          Comment
   103→        </Button>
   104→      </div>
   105→    </div>
   106→  )
   107→}

   212→export const REACTION_EMOJI = ["👍", "❤️", "😂", "🎉", "✅", "👀"] as const

   214→function groupReactions(reactions: CommentReactionDTO[]) {
   215→  const groups = new Map<string, { emoji: string; count: number; names: string[]; mine: boolean }>()
   216→  for (const r of reactions) {
   217→    const g = groups.get(r.emoji) ?? { emoji: r.emoji, count: 0, names: [], mine: false }
   218→    g.count += 1
   219→    g.names.push(r.userName)
   220→    groups.set(r.emoji, g)
   221→  }
   222→  return [...groups.values()]
   223→}

   225→function ReactionRow({
   226→  reactions,
   227→  meId,
   228→  busy,
   229→  onToggle,
   230→}: {
   231→  reactions: CommentReactionDTO[]
   232→  meId: string | null
   233→  busy: boolean
   234→  onToggle: (emoji: string) => void
   235→}) {
   236→  const [picking, setPicking] = React.useState(false)
   237→  const groups = React.useMemo(() => {
   238→    const g = groupReactions(reactions)
   239→    for (const item of g) item.mine = meId != null && reactions.some((r) => r.emoji === item.emoji && r.userId === meId)
   240→    return g
   241→  }, [reactions, meId])
   242→
   243→  React.useEffect(() => {
   244→    if (!picking) return
   245→    const close = (e: MouseEvent) => {
   246→      if (!(e.target as HTMLElement).closest?.("[data-reaction-picker]")) setPicking(false)
   247→    }
   248→    const onKey = (e: KeyboardEvent) => {
   249→      if (e.key === "Escape") setPicking(false)
   250→    }
   251→    window.addEventListener("mousedown", close)
   252→    window.addEventListener("keydown", onKey)
   253→    return () => {
   254→      window.removeEventListener("mousedown", close)
   255→      window.removeEventListener("keydown", onKey)
   256→    }
   257→  }, [picking])
   258→
   259→  return (
   260→    <div className="mt-1.5 flex flex-wrap items-center gap-1 pl-1" data-reactions>
   261→      {groups.map((g) => (
   262→        <Tooltip key={g.emoji}>
   263→          <TooltipTrigger asChild>
   264→            <button
   265→              type="button"
   266→              aria-label={`${g.emoji} reaction, ${g.count}. ${g.mine ? "Click to remove yours" : "Click to react"}`}
   267→              aria-pressed={g.mine}
   268→              disabled={busy}
   269→              onClick={() => onToggle(g.emoji)}
   270→              className={cn(
   271→                "animate-in zoom-in-75 flex h-6 items-center gap-1 rounded-md border px-1.5 text-xs shadow-sm transition-all duration-150 active:scale-90 disabled:opacity-50",
   272→                g.mine
   273→                  ? "border-primary/40 bg-primary/10 text-foreground"
   274→                  : "border-border bg-background text-muted-foreground hover:border-border/80 hover:bg-accent hover:text-foreground"
   275→              )}
   276→            >
   277→              <span aria-hidden>{g.emoji}</span>
   278→              {g.count > 1 && <span className="tnum text-[10px] leading-none">{g.count}</span>}
   279→            </button>
   280→          </TooltipTrigger>
   281→          <TooltipContent side="bottom" className="text-xs">
   282→            {g.names.join(", ")} reacted with {g.emoji}
   283→          </TooltipContent>
   284→        </Tooltip>
   285→      ))}
   286→
   287→      <div className="relative" data-reaction-picker>
   288→        <button
   289→          type="button"
   290→          aria-label="Add reaction"
   291→          aria-expanded={picking}
   292→          disabled={busy}
   293→          onClick={() => setPicking((p) => !p)}
   294→          className="flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground/50 transition-all hover:bg-accent hover:text-foreground active:scale-90 disabled:opacity-50"
   295→        >
   296→          <SmilePlus className={cn("h-3.5 w-3.5 transition-colors", picking && "text-primary")} strokeWidth={1.75} />
   297→        </button>
   298→        {picking && (
   299→          <div
   300→            role="menu"
   301→            aria-label="Pick a reaction"
   302→            className="elev-2 animate-in fade-in-0 zoom-in-95 absolute bottom-8 left-0 z-10 flex origin-bottom-left items-center gap-0.5 rounded-lg border bg-background/95 p-1 backdrop-blur-md duration-150"
   303→          >
   304→            {REACTION_EMOJI.map((emoji) => (
   305→              <button
   306→                key={emoji}
   307→                type="button"
   308→                role="menuitem"
   309→                aria-label={`React with ${emoji}`}
   310→                onClick={() => {
   311→                  onToggle(emoji)
   312→                  setPicking(false)
   313→                }}
   314→                className="flex h-7 w-7 items-center justify-center rounded-md text-sm transition-all hover:scale-125 hover:bg-accent active:scale-90"
   315→              >
   316→                {emoji}
   317→              </button>
   318→            ))}
   319→          </div>
   320→        )}
   321→      </div>
   322→    </div>
   323→  )
   324→}

   358→interface ThreadCardProps {
   359→  comment: CommentDTO
   360→  active: boolean
   361→  busy: boolean
   362→  meId: string | null
   363→  onReply: (text: string) => void
   364→  onToggleResolve: () => void
   365→  onDelete: () => void

   371→function ThreadCard({ comment, active, busy, meId, onReply, onToggleResolve, onDelete, onFocusClick, onEdit, onToggleReaction }: ThreadCardProps) {
   372→  const [replying, setReplying] = React.useState(false)
   373→  const [editing, setEditing] = React.useState<string | null>(null)
   374→  const ref = React.useRef<HTMLDivElement | null>(null)
   375→
   376→  React.useEffect(() => {
   377→    if (active) ref.current?.scrollIntoView({ behavior: "smooth", block: "nearest" })
   378→  }, [active])
   379→
   380→  const replies = comment.replies ?? []
   381→  const lastActivity = replies.length > 0 ? replies[replies.length - 1].createdAt : comment.createdAt
   382→
   383→  return (
   384→    <div
   385→      ref={ref}
   386→      data-comment-id={comment.id}
   387→      className={cn(
   388→        "animate-fade-in scroll-mt-4 rounded-lg border bg-card p-3 transition-[border-color,box-shadow,opacity] group/thread",
   389→        active && "border-primary/60 ring-2 ring-primary/20",
   390→        comment.resolved && "opacity-75"
   391→      )}
   392→    >
   393→      <div className="flex gap-2.5">
   394→        <CommentAvatar name={comment.authorName} color={comment.authorColor} />
   395→        {editing === comment.id ? (
   396→          <div className="min-w-0 flex-1">
   397→            <EditForm
   398→              initial={comment.content}
   399→              busy={busy}
   400→              onSubmit={(text) => {
   401→                onEdit?.(comment.id, text)
   402→                setEditing(null)
   403→              }}
   404→              onCancel={() => setEditing(null)}
   405→            />
   406→          </div>
   407→        ) : (
   408→          <MessageBody authorName={comment.authorName} authorColor={comment.authorColor} time={lastActivity} resolved={comment.resolved}>
   409→            {comment.content}
   410→          </MessageBody>
   411→        )}
   412→      </div>
   413→
   414→      {comment.quote && (
   415→        <button
   416→          onClick={onFocusClick}
   417→          className="mt-2 flex w-full items-start gap-2 rounded-lg bg-muted/70 p-2 text-left transition-colors hover:bg-muted"
   418→          aria-label="Show quoted text in document"
   419→        >
   420→          <Quote className="mt-0.5 h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   421→          <p className="line-clamp-2 text-xs italic leading-snug text-muted-foreground">{comment.quote}</p>
   422→        </button>
   423→      )}
   424→
   425→      {onToggleReaction && (
   426→        <ReactionRow
   427→          reactions={comment.reactions ?? []}
   428→          meId={meId}
   429→          busy={busy}
   430→          onToggle={(emoji) => onToggleReaction(comment.id, emoji)}
   431→        />
   432→      )}
   433→
   434→      {replies.length > 0 && (
   435→        <div className="mt-2 space-y-2 border-l border-border pl-3 ml-4">
   436→          {replies.map((r) => (
   437→            <div key={r.id} className="space-y-0.5">
   438→              <div className="flex gap-2.5">
   439→                <CornerDownRight className="mt-1 h-3.5 w-3.5 shrink-0 text-muted-foreground/60" />
   440→                <div className="flex gap-2">
   441→                  <CommentAvatar name={r.authorName} color={r.authorColor} size="sm" />
   442→                  {editing === r.id ? (
   443→                    <div className="min-w-0 flex-1">
   444→                      <EditForm
   445→                        initial={r.content}
   446→                        busy={busy}
   447→                        onSubmit={(text) => {
   448→                          onEdit?.(r.id, text)
   449→                          setEditing(null)
   450→                        }}
   451→                        onCancel={() => setEditing(null)}
   452→                      />
   453→                    </div>
   454→                  ) : (
   455→                    <MessageBody authorName={r.authorName} authorColor={r.authorColor} time={r.createdAt} resolved={r.resolved}>
   456→                      {r.content}
   457→                    </MessageBody>
   458→                  )}
   459→                </div>
   460→                {editing !== r.id && (meId === r.authorId || meId === null) && onEdit && (
   461→                  <Button
   462→                    variant="ghost"
   463→                    size="icon"
   464→                    aria-label="Edit reply"
   465→                    className="h-6 w-6 shrink-0 rounded-md text-muted-foreground/60 opacity-0 transition-opacity hover:text-foreground group-hover/thread:opacity-100 focus-visible:opacity-100"
   466→                    onClick={() => setEditing(r.id)}
   467→                    disabled={busy}

   469→                    <Pencil className="h-3 w-3" />
   470→                  </Button>
   471→                )}
   472→              </div>
   473→              {onToggleReaction && (
   474→                <div
   475→                  className={cn(
   476→                    "ml-6 transition-opacity",
   477→                    (r.reactions?.length ?? 0) === 0 &&
   478→                      "opacity-0 focus-within:opacity-100 group-hover/thread:opacity-100"
   479→                  )}
   480→                >
   481→                  <ReactionRow
   482→                    reactions={r.reactions ?? []}
   483→                    meId={meId}
   484→                    busy={busy}
   485→                    onToggle={(emoji) => onToggleReaction(r.id, emoji)}
   486→                  />
   487→                </div>
   488→              )}
   489→            </div>
   490→          ))}
   491→        </div>
   492→      )}
   493→
   494→      {replying ? (
   495→        <ReplyForm
   496→          onSubmit={(text) => {
   497→            onReply(text)
   498→            setReplying(false)
   499→          }}
   500→          onCancel={() => setReplying(false)}
   501→        />
   502→      ) : (
   503→        <div className="mt-2 flex items-center gap-1 pl-1">
   504→          <Button
   505→            variant="ghost"
   506→            size="sm"
   507→            className="h-7 gap-1 rounded-md px-2.5 text-xs text-muted-foreground hover:text-foreground"
   508→            onClick={() => setReplying(true)}
   509→            disabled={busy}
   510→          >
   511→            <CornerDownRight className="h-3.5 w-3.5" /> Reply
   512→          </Button>
   513→          {(meId === comment.authorId || meId === null) && onEdit && editing !== comment.id && (
   514→            <Button
   515→              variant="ghost"
   516→              size="sm"
   517→              className="h-7 gap-1 rounded-md px-2.5 text-xs text-muted-foreground hover:text-foreground"
   518→              onClick={() => setEditing(comment.id)}
   519→              disabled={busy}
   520→            >
   521→              <Pencil className="h-3.5 w-3.5" /> Edit
   522→            </Button>
   523→          )}
   524→          <Tooltip>
   525→            <TooltipTrigger asChild>
   526→              <Button
   527→                variant="ghost"
   528→                size="sm"
   529→                className="h-7 gap-1 rounded-md px-2.5 text-xs text-muted-foreground hover:text-foreground"
   530→                onClick={onToggleResolve}
   531→                disabled={busy}
   532→              >
   533→                {comment.resolved ? <RotateCcw className="h-3.5 w-3.5" /> : <Check className="h-3.5 w-3.5" />}
   534→                {comment.resolved ? "Reopen" : "Resolve"}

   537→            <TooltipContent side="bottom" className="text-xs">
   538→              {comment.resolved ? "Reopen this thread" : "Mark as resolved"}
   539→            </TooltipContent>
   540→          </Tooltip>
   541→          {(meId === comment.authorId || meId === null) && (
   542→            <Tooltip>
   543→              <TooltipTrigger asChild>
   544→                <Button
   545→                  variant="ghost"
   546→                  size="sm"
   547→                  aria-label="Delete comment thread"
   548→                  className="ml-auto h-7 gap-1 rounded-md px-2.5 text-xs text-muted-foreground hover:text-destructive"
   549→                  onClick={onDelete}
   550→                  disabled={busy}
   551→                >
   552→                  {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Trash2 className="h-3.5 w-3.5" />}
   553→                </Button>
   554→              </TooltipTrigger>
   555→              <TooltipContent side="bottom" className="text-xs">Delete thread</TooltipContent>
   556→            </Tooltip>
   557→          )}
   558→        </div>
   559→      )}
   560→    </div>
   561→  )
   562→}

   564→/* ------------------------------------------------------------- suggestion cards */
   565→
   566→function SuggestionCard({
   567→  s,
   568→  active,
   569→  onAccept,
   570→  onReject,
   571→  onFocus,
   572→}: {
   573→  s: SuggestionInfo
   574→  active: boolean
   575→  onAccept: (sid: string) => void
   576→  onReject: (sid: string) => void
   577→  onFocus: (sid: string) => void
   578→}) {
   579→  const verb =
   580→    s.kind === "del" ? "deleted" : s.kind === "para" ? "suggested a paragraph break" : "inserted"
   581→  const preview =
   582→    s.kind === "para" ? "¶" : s.text.length > 120 ? s.text.slice(0, 120).trimEnd() + "…" : s.text
   583→
   584→  return (
   585→    <div
   586→      data-sug-card={s.sid}
   587→      className={cn(
   588→        "animate-fade-in rounded-lg border bg-card p-3 transition-[border-color,box-shadow] group/sug",
   589→        active ? "border-primary/60 ring-2 ring-primary/20" : "hover:border-border/80"
   590→      )}
   591→    >
   592→      <div className="flex gap-2.5">
   593→        <CommentAvatar name={s.authorName} color={s.authorColor} size="sm" />
   594→        <div className="min-w-0 flex-1">
   595→          <p className="text-xs leading-snug">
   596→            <button
   597→              className="font-semibold text-foreground hover:underline"
   598→              onClick={() => onFocus(s.sid)}
   599→              title="Show in document"
   600→            >
   601→              {s.authorName}
   602→            </button>{" "}
   603→            <span className="text-muted-foreground">{verb}</span>
   604→            {s.createdAt > 0 && (
   605→              <span className="text-muted-foreground"> · {relativeTime(new Date(s.createdAt).toISOString())}</span>
   606→            )}
   607→          </p>
   608→          {s.kind !== "para" && (
   609→            <button
   610→              className="mt-1.5 block w-full rounded-md bg-muted/60 px-2 py-1.5 text-left text-xs leading-snug transition-colors hover:bg-muted"
   611→              onClick={() => onFocus(s.sid)}
   612→              aria-label="Show this suggestion in the document"
   613→            >
   614→              <span
   615→                className="line-clamp-3"
   616→                style={{
   617→                  textDecorationLine: s.kind === "del" ? "line-through" : "underline",
   618→                  textDecorationColor: s.authorColor,
   619→                  textDecorationThickness: "1.5px",
   620→                  textUnderlineOffset: "2px",
   621→                  color: s.kind === "del" ? s.authorColor : undefined,
   622→                }}
   623→              >
   624→                {preview || "(empty)"}
   625→              </span>
   626→            </button>
   627→          )}
   628→        </div>
   629→      </div>
   630→      <div className="mt-2 flex items-center justify-end gap-1.5">
   631→        <Button
   632→          variant="ghost"
   633→          size="sm"
   634→          className="h-7 gap-1.5 rounded-md px-2.5 text-xs text-muted-foreground hover:text-foreground"
   635→          onClick={() => onReject(s.sid)}
   636→        >
   637→          <X className="h-3.5 w-3.5" /> Reject
   638→        </Button>
   639→        <Button
   640→          size="sm"
   641→          className="h-7 gap-1.5 rounded-md px-2.5 text-xs"
   642→          onClick={() => onAccept(s.sid)}
   643→        >
   644→          <Check className="h-3.5 w-3.5" /> Accept
   645→        </Button>
   646→      </div>
   647→    </div>
   648→  )
   649→}

   690→  return (

   693→      data-open={open}

   710→            </span>
   711→          )}

   713→            variant="ghost"
   714→            size="icon"
   715→            aria-label="Close comments panel"
   716→            className="ml-auto h-8 w-8 rounded-md text-muted-foreground"
   717→            onClick={onClose}
   718→          >
   719→            <X className="h-4 w-4" />
   720→          </Button>
   721→        </div>
   722→

   724→        <div className="slim-scroll flex-1 space-y-3 overflow-y-auto p-3">
   725→          {suggestions.length > 0 && (
   726→            <div className="space-y-2">
   727→              <div className="sticky top-0 z-[1] -mx-1 flex items-center gap-2 bg-background/95 px-1 py-1 backdrop-blur-sm">
   728→                <PencilLine className="h-3.5 w-3.5 text-primary" />

   730→                  Suggestions
   731→                </h3>
   732→                <span className="rounded-md bg-primary/10 px-1.5 py-0.5 text-[11px] font-medium tabular-nums text-primary">
   733→                  {suggestions.length}

   740→                  active={activeSuggestionId === s.sid}

   817→                </div>
   818→              )}
   819→            </>
   820→          )}
   821→        </div>
   822→      </div>
