    58→/* ------------------------------ theme mode hook ----------------------------- */
    59→
    60→type ThemeMode = "light" | "dark" | "system"
    61→
    62→function systemPreference(): "light" | "dark" {
    63→  if (typeof window === "undefined" || !window.matchMedia) return "light"
    64→  return window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light"
    65→}
    66→
    67→/**
    68→ * Light / Dark / System selection on top of next-themes' setTheme. The app's
    69→ * ThemeProvider has system resolution disabled, so "System" is resolved here
    70→ * via matchMedia and applied as an explicit light/dark theme — and followed
    71→ * live if the OS preference changes. The chosen mode is mirrored to
    72→ * localStorage and re-synced when the theme is toggled elsewhere (header).
    73→ */
    74→function useThemeMode(): { mode: ThemeMode; setMode: (mode: ThemeMode) => void } {
    75→  const { theme, setTheme } = useTheme()
    76→  const [mode, setModeState] = React.useState<ThemeMode>(() => {
    77→    if (typeof window === "undefined") return "light"
    78→    let own: string | null = null
    79→    try {
    80→      own = window.localStorage.getItem(THEME_MODE_STORAGE_KEY)
    81→    } catch {
    82→      // storage unavailable
    83→    }
    84→    if (own === "light" || own === "dark" || own === "system") {
    85→      const applied = own === "system" ? systemPreference() : own
    86→      if ((theme === "light" || theme === "dark") && theme !== applied) return theme
    87→      return own
    88→    }
    89→    return theme === "dark" ? "dark" : "light"
    90→  })
    91→
    92→  const setMode = React.useCallback(
    93→    (next: ThemeMode) => {
    94→      setModeState(next)
    95→      try {
    96→        window.localStorage.setItem(THEME_MODE_STORAGE_KEY, next)
    97→      } catch {
    98→        // storage unavailable — still apply for this session
    99→      }
   100→      setTheme(next === "system" ? systemPreference() : next)
   101→    },
   102→    [setTheme]
   103→  )
   104→
   105→  // keep the card selection honest when the theme is toggled outside Settings
   106→  React.useEffect(() => {
   107→    if (theme !== "light" && theme !== "dark") return
   108→    const applied = mode === "system" ? systemPreference() : mode
   109→    if (theme !== applied) setModeState(theme)
   110→  }, [theme, mode])
   111→
   112→  // follow OS preference changes while in System mode
   113→  React.useEffect(() => {
   114→    if (mode !== "system") return
   115→    const mq = window.matchMedia("(prefers-color-scheme: dark)")
   116→    const onChange = () => setTheme(mq.matches ? "dark" : "light")
   117→    mq.addEventListener("change", onChange)
   118→    return () => mq.removeEventListener("change", onChange)
   119→  }, [mode, setTheme])
   120→
   121→  return { mode, setMode }
   122→}

   156→function formatBytes(bytes: number): string {
   157→  if (bytes >= 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`

   493→function DataSection() {
   494→  const { toast } = useToast()
   495→  const [counts, setCounts] = React.useState<NonNullable<ExportPayload["counts"]> | null>(null)
   496→  const [bytes, setBytes] = React.useState(0)
   497→  const [error, setError] = React.useState<string | null>(null)
   498→
   499→  React.useEffect(() => {
   500→    let live = true
   501→    const load = async () => {
   502→      setError(null)
   503→      try {
   504→        const res = await fetch("/api/export")
   505→        if (!res.ok) throw new Error(`Failed to load storage stats (${res.status})`)
   506→        const text = await res.text()
   507→        if (!live) return
   508→        const payload = JSON.parse(text) as ExportPayload
   509→        setCounts(payload.counts ?? {})
   510→        setBytes(new TextEncoder().encode(text).length)
   511→      } catch (e) {
   512→        if (!live) return
   513→        setError(e instanceof Error ? e.message : "Something went wrong")
   514→      }
   515→    }
   516→    void load()
   517→    return () => {
   518→      live = false
   519→    }
   520→  }, [])
   521→
   522→  const exportAll = () => {
   523→    const a = document.createElement("a")
   524→    a.href = "/api/export"
   525→    a.setAttribute("download", "")
   526→    document.body.appendChild(a)
   527→    a.click()
   528→    a.remove()
   529→    toast({ title: "Download started", description: "Your workspace is exporting as JSON." })
   530→  }
   531→
   532→  const resetLocal = () => {
   533→    for (const key of [
   534→      USER_STORAGE_KEY,
   535→      ACCENT_STORAGE_KEY,
   536→      FONT_STORAGE_KEY,
   537→      ZOOM_STORAGE_KEY,
   538→      THEME_MODE_STORAGE_KEY,
   539→    ]) {
   540→      try {
   541→        window.localStorage.removeItem(key)
   542→      } catch {
   543→        // ignore
   544→      }
   545→    }
   546→    window.location.reload()
   547→  }
   548→
   549→  const usedPercent = counts ? Math.min(100, (bytes / STORAGE_QUOTA_BYTES) * 100) : 0
   550→
   551→  return (
   552→    <SectionCard
   553→      title="Data & storage"
   554→      description="Everything you create lives in your local Z workspace database."
   555→    >
   556→      {error ? (
   557→        <p className="rounded-lg border border-destructive/30 bg-destructive/5 px-4 py-3 text-[13px] text-destructive">
   558→          {error}
   559→        </p>
   560→      ) : counts ? (
   561→        <>
   562→          <div>
   563→            <div className="flex items-baseline justify-between gap-3">
   564→              <p className="text-[13px] font-medium">Workspace storage</p>
   565→              <p className="text-xs text-muted-foreground">
   566→                <span className="tnum font-medium text-foreground">{formatBytes(bytes)}</span> used
   567→                <span className="opacity-70"> · 10 MB quota</span>
   568→              </p>
   569→            </div>
   570→            <div
