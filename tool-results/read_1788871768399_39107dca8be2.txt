    39→function titleFor(filter: string, folders: FolderDTO[], activeFolderId: string | null) {
    40→  if (filter === "trash") return "Trash"
    41→  if (filter === "starred") return "Starred"
    42→  if (filter === "folder") {
    43→    const f = folders.find((x) => x.id === activeFolderId)
    44→    return f ? f.name : "Folder"
    45→  }
    46→  return "Recent documents"
    47→}

   472→/** "Add tags" submenu: checkbox per tag + inline "New tag…" creation (applied to this doc) */
   473→function AddTagsSubmenu({ doc }: { doc: DocumentMeta }) {
   474→  const tags = useDocsStore((s) => s.tags)
   475→  const setDocTags = useDocsStore((s) => s.setDocTags)
   476→  const createTag = useDocsStore((s) => s.createTag)
   477→  const { toast } = useToast()
   478→  const { t } = useI18n()
   479→  const [creating, setCreating] = React.useState(false)
   480→  const [name, setName] = React.useState("")
   481→  const [color, setColor] = React.useState<string>(DEFAULT_TAG_COLOR)
   482→  const docTagIds = new Set((doc.tags ?? []).map((tg) => tg.id))
   483→
   484→  const toggleTag = (tagId: string) => {
   485→    const ids = (doc.tags ?? []).map((tg) => tg.id)
   486→    void setDocTags(doc.id, ids.includes(tagId) ? ids.filter((i) => i !== tagId) : [...ids, tagId])
   487→  }
   488→
   489→  const confirmCreate = async () => {
   490→    const trimmed = name.trim()
   491→    if (!trimmed) return

   529→            onBlur={(e) => {
   530→              // cancel only when focus leaves the whole creation form (palette clicks stay inside)

   963→function BulkTagDialog({
   964→  open,

   966→  count,
   967→  busy,
   968→  onApply,
   969→}: {
   970→  open: boolean
   971→  onOpenChange: (open: boolean) => void
   972→  count: number
   973→  busy: boolean
   974→  onApply: (tagIds: string[]) => void

   980→
   981→  const [selected, setSelected] = React.useState<string[]>([])
   982→  const [newName, setNewName] = React.useState("")
   983→  const [newColor, setNewColor] = React.useState(DEFAULT_TAG_COLOR)
   984→  const [creating, setCreating] = React.useState(false)
   985→
   986→  // reset the draft state whenever the dialog (re)opens
   987→  React.useEffect(() => {
   988→    if (open) {
   989→      setSelected([])
   990→      setNewName("")
   991→      setNewColor(DEFAULT_TAG_COLOR)

   993→  }, [open])
   994→

   996→    setSelected((cur) => (cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id]))
   997→

   999→    const name = newName.trim()
  1000→    if (!name || creating) return
  1001→    setCreating(true)
  1002→    const tag = await createTag(name, newColor)
  1003→    setCreating(false)
  1004→    if (!tag) {
  1005→      toast({ title: t("Couldn’t create the tag"), variant: "destructive" })
  1006→      return
  1007→    }
  1008→    setNewName("")
  1009→    setNewColor(DEFAULT_TAG_COLOR)
  1010→    setSelected((cur) => (cur.includes(tag.id) ? cur : [...cur, tag.id]))
  1011→  }
