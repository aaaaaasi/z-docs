    60→const TXT_BLOCK_TAGS = new Set(["p", "div", "li", "blockquote", "h1", "h2", "h3", "h4", "h5", "h6", "pre", "tr", "table", "figure"])
    61→
    62→/** Client-side DOM walk producing a genuinely structured .txt export:
    63→ *  underlined headings, real list markers with nesting, quote prefixes,
    64→ *  verbatim code blocks, tables as `a | b` rows, image alt placeholders. */
    65→function htmlToPlainText(html: string): string {
    66→  if (!html) return ""
    67→  const doc = new DOMParser().parseFromString(html, "text/html")
    68→  const lines: string[] = []
    69→  const push = (s: string) => {
    70→    if (s.trim()) lines.push(s.replace(/\u00a0/g, " ").trimEnd())
    71→  }
    72→
    73→  const inlineText = (node: Node): string => {
    74→    let s = ""
    75→    node.childNodes.forEach((ch) => {
    76→      if (ch.nodeType === Node.TEXT_NODE) s += (ch.textContent || "").replace(/\s+/g, " ")
    77→      else if (ch.nodeType === Node.ELEMENT_NODE) {
    78→        const el = ch as Element
    79→        const tag = el.tagName.toLowerCase()
    80→        if (tag === "br") s += "\n"
    81→        else if (tag === "img") {
    82→          const alt = el.getAttribute("alt")
    83→          if (alt) s += `[${alt}]`
    84→        } else if (TXT_BLOCK_TAGS.has(tag)) s += `\n${inlineText(el)}\n`
    85→        else s += inlineText(el)
    86→      }
    87→    })
    88→    return s
    89→  }
    90→
    91→  const walkList = (list: Element, depth: number) => {
    92→    const ordered = list.tagName.toLowerCase() === "ol"
    93→    const indent = "  ".repeat(depth)
    94→    let i = 1
    95→    list.childNodes.forEach((n) => {
    96→      if (n.nodeType !== Node.ELEMENT_NODE) return
    97→      const li = n as Element
    98→      if (li.tagName.toLowerCase() !== "li") return
    99→      let inline = ""
   100→      const nested: Element[] = []
   101→      li.childNodes.forEach((ch) => {
   102→        if (ch.nodeType === Node.TEXT_NODE) inline += (ch.textContent || "").replace(/\s+/g, " ")
   103→        else if (ch.nodeType === Node.ELEMENT_NODE) {
   104→          const cel = ch as Element
   105→          const ctag = cel.tagName.toLowerCase()
   106→          if (ctag === "ul" || ctag === "ol") nested.push(cel)
   107→          else if (ctag === "br") inline += "\n"
   108→          else inline += inlineText(cel)
   109→        }
   110→      })
   111→      const marker = ordered ? `${i++}. ` : "- "
   112→      push(indent + marker + inline.trim())
   113→      nested.forEach((nl) => walkList(nl, depth + 1))
   114→    })
   115→  }
   116→
   117→  const walkBlocks = (parent: Element, depth = 0) => {
   118→    const indent = "  ".repeat(depth)
   119→    parent.childNodes.forEach((node) => {
   120→      if (node.nodeType === Node.TEXT_NODE) {
   121→        push(indent + (node.textContent || "").replace(/\s+/g, " ").trim())
   122→        return
   123→      }
   124→      if (node.nodeType !== Node.ELEMENT_NODE) return
   125→      const el = node as Element
   126→      const tag = el.tagName.toLowerCase()
   127→      if (/^h[1-6]$/.test(tag)) {
   128→        const text = inlineText(el).trim()
   129→        if (!text) return
   130→        lines.push("")
   131→        push(indent + text)
   132→        if (tag === "h1") push("=".repeat(Math.min(60, Math.max(6, text.length * 2))))
   133→        else if (tag === "h2") push("-".repeat(Math.min(60, Math.max(6, text.length * 2))))
   134→        lines.push("")
   135→      } else if (tag === "p") {
   136→        const text = inlineText(el).replace(/[ \t]+\n/g, "\n").trim()
   137→        if (text) text.split("\n").forEach((l) => push(indent + l))
   138→      } else if (tag === "ul" || tag === "ol") {
   139→        walkList(el, depth)
   140→      } else if (tag === "blockquote") {
   141→        const inner = inlineText(el).trim()
   142→        if (inner) {
   143→          lines.push("")
   144→          inner.split("\n").forEach((l) => push("> " + l.trim()))
   145→          lines.push("")
   146→        }
   147→      } else if (tag === "pre") {
   148→        const text = el.textContent || ""
   149→        lines.push("")
   150→        text.split("\n").forEach((l) => lines.push(l))
   151→        lines.push("")
   152→      } else if (tag === "hr") {
   153→        lines.push("")
   154→        push("-".repeat(40))
   155→        lines.push("")
   156→      } else if (tag === "table") {
   157→        lines.push("")
   158→        Array.from(el.querySelectorAll("tr")).forEach((tr) => {
   159→          const cells = Array.from(tr.querySelectorAll("th,td")).map((td) => inlineText(td).replace(/\s+/g, " ").trim())
   160→          push(cells.join(" | "))
   161→        })
   162→        lines.push("")
   163→      } else if (tag === "br") {
   164→        lines.push("")
   165→      } else {
   166→        walkBlocks(el, depth)
   167→      }
   168→    })
   169→  }
   170→
   171→  walkBlocks(doc.body)
   172→  return lines.join("\n").replace(/\n{3,}/g, "\n\n").trim() + "\n"
   173→}

   603→        case "insertText": {

   607→          // let the default insertText happen (it stays inside the mark)

   627→          const dir = input.inputType === "deleteContentBackward" ? "backward" : "forward"

  1046→        }

  1048→        case "col-right": {
  1049→          const cells = insertTableColumn(ctx, op === "col-left" ? "left" : "right")

  1059→            if (nextRow?.cells[0]) placeCaretInCell(nextRow.cells[0])

  1062→          break
  1063→        }
  1064→        case "delete-col": {
  1065→          const removedTable = !deleteTableColumn(ctx)
  1066→          if (!removedTable) {
  1067→            const row = ctx.table.rows[ctx.rowIndex]
  1068→            const cell = row?.cells[Math.min(ctx.colIndex, row.cells.length - 1)]
  1069→            if (cell) placeCaretInCell(cell)
  1070→          }

  1072→          break
