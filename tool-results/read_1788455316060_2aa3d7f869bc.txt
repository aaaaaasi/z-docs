    61→  const [menuTableInfo, setMenuTableInfo] = React.useState<TableInfo | null>(null)

    69→  /* ---------------- comments state ---------------- */

    71→  const [commentsLoading, setCommentsLoading] = React.useState(false)
    72→  const [commentsOpen, setCommentsOpen] = React.useState(false)
    73→  const [activeCommentId, setActiveCommentId] = React.useState<string | null>(null)
    74→  const [commentBusy, setCommentBusy] = React.useState(false)
    75→  const [contentTick, setContentTick] = React.useState(0)
    76→  const [bubble, setBubble] = React.useState<{ x: number; y: number } | null>(null)
    77→  const [pendingQuote, setPendingQuoteState] = React.useState<PendingQuote | null>(null)
    78→  const pendingQuoteRef = React.useRef<PendingQuote | null>(null)

    80→  /* ---------------- refs ---------------- */
    81→  const pageRef = React.useRef<HTMLDivElement | null>(null)

    83→  const savedRangeRef = React.useRef<Range | null>(null)

    85→  const contentRef = React.useRef("")
    86→  const latestTitleRef = React.useRef("")
    87→  const dirtyRef = React.useRef(false)
    88→  const lastInputAtRef = React.useRef(0)
    89→  const lastCursorEmitRef = React.useRef(0)
    90→  const saveTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null)
    91→  const statsTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null)
    92→  const broadcastTimerRef = React.useRef<ReturnType<typeof setTimeout> | null>(null)
    93→  const versionCounterRef = React.useRef(0)
    94→  const myIdRef = React.useRef<string | null>(null)
    95→
    96→  /* ---------------- load document ---------------- */
    97→  React.useEffect(() => {
    98→    let cancelled = false
    99→    setDoc(null)
   100→    setLoadError(null)
   101→    setSaveStatus("saved")
   102→    setRemoteContent(null)
   103→    fetch(`/api/documents/${docId}`)
   104→      .then((r) => {
   105→        if (r.status === 404) throw new Error("not-found")
   106→        if (!r.ok) throw new Error("load-failed")
   107→        return r.json() as Promise<{ document: DocumentDTO }>
   108→      })
   109→      .then((data) => {
   110→        if (cancelled) return
   111→        const d = data.document
   112→        setDoc(d)
   113→        setTitle(d.title)
   114→        latestTitleRef.current = d.title
   115→        contentRef.current = d.content
   116→        setStarred(d.starred)
   117→        setStats(docStats(d.content))
   118→        setLastSavedAt(new Date(d.updatedAt))
   119→      })
   120→      .catch((e: Error) => {
   121→        if (cancelled) return
   122→        setLoadError(e.message === "not-found" ? "This document doesn't exist anymore." : "Couldn't load this document.")
   123→      })
   124→    return () => {
   125→      cancelled = true
   126→    }
   127→  }, [docId])
   128→
   129→  // auto-open the AI writer when requested from the template gallery
   130→  React.useEffect(() => {
   131→    if (openAiOnEditor && doc) {
   132→      setDialog("helpwrite")
   133→      setOpenAiOnEditor(false)
   134→    }
   135→  }, [openAiOnEditor, doc, setOpenAiOnEditor])
   136→
   137→  /* ---------------- comments: load & helpers ---------------- */
   138→  React.useEffect(() => {
   139→    if (!docId) return
   140→    let cancelled = false
   141→    setComments([])
   142→    setCommentsLoading(true)
   143→    setActiveCommentId(null)
   144→    fetch(`/api/documents/${docId}/comments`)
   145→      .then((r) => (r.ok ? (r.json() as Promise<{ comments: CommentDTO[] }>) : Promise.reject(new Error("failed"))))
   146→      .then((data) => {
   147→        if (!cancelled) setComments(data.comments ?? [])
   148→      })
   149→      .catch(() => {
   150→        // non-fatal: the sidebar shows an empty state
   151→      })
   152→      .finally(() => {
   153→        if (!cancelled) setCommentsLoading(false)
   154→      })
   155→    return () => {
   156→      cancelled = true
   157→    }
   158→  }, [docId])
   159→
   160→  const setPendingQuote = React.useCallback((pq: PendingQuote | null) => {
   161→    pendingQuoteRef.current = pq
   162→    setPendingQuoteState(pq)
   163→  }, [])
   164→
   165→  const refreshComments = React.useCallback(async () => {
   166→    if (!docId) return
   167→    try {
   168→      const res = await fetch(`/api/documents/${docId}/comments`)
   169→      if (!res.ok) return
   170→      const data = (await res.json()) as { comments: CommentDTO[] }
   171→      setComments(data.comments ?? [])
   172→    } catch {
   173→      // ignore — next mutation will retry
   174→    }
   175→  }, [docId])
   176→
   177→  /* ---------------- collaboration ---------------- */
   178→  const onDocChange = React.useCallback((p: DocChangePayload) => {
   179→    if (myIdRef.current && p.by === myIdRef.current) return
   180→    // last-write-wins: skip remote updates while actively typing
   181→    if (Date.now() - lastInputAtRef.current < 1600) return
   182→    if (p.content != null) {
   183→      if (p.content !== contentRef.current) {
   184→        contentRef.current = p.content
   185→        setRemoteContent(p.content)
   186→        setStats(docStats(p.content))
   187→      }
   188→      setSaveStatus("saved")
   189→      setLastSavedAt(new Date())
   190→    }
   191→    if (p.title != null && document.activeElement !== titleInputRef.current && p.title !== latestTitleRef.current) {
   192→      latestTitleRef.current = p.title
   193→      setTitle(p.title)
   194→    }
   195→  }, [])
   196→
   197→  const onCommentsChanged = React.useCallback(
   198→    (p: CommentsChangedPayload) => {
   199→      void refreshComments()
   200→      if (p.action === "add" || p.action === "reply") {
   201→        toast({ title: "New comment activity", description: "A collaborator updated the discussion." })
   202→      }
   203→    },
   204→    [refreshComments, toast]
   205→  )
   206→
   207→  const { connected, myId, presence, remoteCursors, emitDocChange, emitCursor, emitCommentsChanged } = useCollab(
   208→    docId || null,
   209→    user,
   210→    onDocChange,
   211→    onCommentsChanged
   212→  )
   213→  React.useEffect(() => {
   214→    myIdRef.current = myId
   215→  }, [myId])
   216→
   217→  /* ---------------- autosave ---------------- */
   218→  const performSave = React.useCallback(
   219→    async (manual = false) => {
   220→      if (!docId) return
   221→      const content = contentRef.current
   222→      const t = latestTitleRef.current
   223→      setSaveStatus("saving")
   224→      try {
   225→        const res = await fetch(`/api/documents/${docId}`, {
   226→          method: "PATCH",
   227→          headers: { "Content-Type": "application/json" },
   228→          body: JSON.stringify({ content, title: t }),
   229→        })
   230→        if (!res.ok) throw new Error("save failed")
   231→        const data = (await res.json()) as { document: DocumentDTO }
   232→        dirtyRef.current = false
   233→        setSaveStatus("saved")
   234→        setLastSavedAt(new Date())
   235→        patchDocMeta(docId, {
   236→          title: t,
   237→          snippet: getSnippet(content),
   238→          wordCount: countWords(htmlToText(content)),
   239→          updatedAt: data.document.updatedAt,
   240→        })
   241→        if (manual) toast({ title: "Saved", description: "All changes are safe." })
   242→      } catch {
   243→        setSaveStatus("error")
   244→        // retry shortly
   245→        if (saveTimerRef.current) clearTimeout(saveTimerRef.current)
   246→        saveTimerRef.current = setTimeout(() => void performSave(), 5000)
   247→      }
   248→    },
   249→    [docId, patchDocMeta, toast]
   250→  )
   251→
   252→  const scheduleSave = React.useCallback(() => {
   253→    dirtyRef.current = true
   254→    setSaveStatus("unsaved")
   255→    if (saveTimerRef.current) clearTimeout(saveTimerRef.current)
   256→    saveTimerRef.current = setTimeout(() => void performSave(), 900)
   257→  }, [performSave])
   258→
   259→  // flush unsaved changes when leaving the doc
   260→  React.useEffect(() => {
   261→    const id = docId
   262→    return () => {
   263→      if (!dirtyRef.current || !id) return
   264→      try {
   265→        fetch(`/api/documents/${id}`, {
   266→          method: "PATCH",
   267→          keepalive: true,
   268→          headers: { "Content-Type": "application/json" },
   269→          body: JSON.stringify({ content: contentRef.current, title: latestTitleRef.current }),
   270→        }).catch(() => {})
   271→      } catch {
   272→        // ignore
   273→      }
   274→    }
   275→  }, [docId])
   276→
   277→  // warn before leaving with unsaved changes
   278→  React.useEffect(() => {
   279→    const onBefore = (e: BeforeUnloadEvent) => {
   280→      if (dirtyRef.current) {
   281→        e.preventDefault()
   282→        e.returnValue = ""
   283→      }
   284→    }
   285→    window.addEventListener("beforeunload", onBefore)
   286→    return () => window.removeEventListener("beforeunload", onBefore)
   287→  }, [])
   288→
   289→  /* ---------------- content input pipeline ---------------- */
   290→  const scheduleStats = React.useCallback(() => {
   291→    if (statsTimerRef.current) clearTimeout(statsTimerRef.current)
   292→    statsTimerRef.current = setTimeout(() => {
   293→      setStats(docStats(contentRef.current))
   294→      // bump so the comment highlight overlays re-align after local edits
   295→      setContentTick((t) => t + 1)
   296→    }, 400)
   297→  }, [])
   298→
   299→  const scheduleBroadcast = React.useCallback(() => {
   300→    if (broadcastTimerRef.current) clearTimeout(broadcastTimerRef.current)
   301→    broadcastTimerRef.current = setTimeout(() => {
   302→      versionCounterRef.current += 1
   303→      emitDocChange(contentRef.current, latestTitleRef.current, versionCounterRef.current)
   304→    }, 300)
   305→  }, [emitDocChange])
   306→
   307→  const handleInput = React.useCallback(() => {
   308→    const el = pageRef.current
   309→    if (!el) return
   310→    contentRef.current = el.innerHTML
   311→    lastInputAtRef.current = Date.now()
   312→    scheduleSave()
   313→    scheduleBroadcast()
   314→    scheduleStats()
   315→  }, [scheduleSave, scheduleBroadcast, scheduleStats])
   316→
   317→  /* ---------------- format state ---------------- */
   318→  const refreshFmt = React.useCallback(() => {
   319→    const q = (c: string) => {
   320→      try {
   321→        return document.queryCommandState(c)
   322→      } catch {
   323→        return false
   324→      }
   325→    }
   326→    let block = "p"
   327→    try {
   328→      block = (document.queryCommandValue("formatBlock") || "p").toLowerCase()
   329→    } catch {
   330→      // ignore
   331→    }
   332→    const sel = window.getSelection()
   333→    const node = sel?.anchorNode
   334→    const elem =
   335→      node && node.nodeType === Node.TEXT_NODE ? node.parentElement : (node as HTMLElement | null)
   336→    const computed = elem ? window.getComputedStyle(elem) : null
   337→    const fontName = (computed?.fontFamily ?? "Arial").split(",")[0].replace(/["']/g, "").trim()
   338→    const fontSizePt = computed ? Math.max(6, Math.round(parseFloat(computed.fontSize) / 1.3333)) : 11
   339→
   340→    setFmt({
   341→      bold: q("bold"),
   342→      italic: q("italic"),
   343→      underline: q("underline"),
   344→      strike: q("strikeThrough"),
   345→      superscript: q("superscript"),
   346→      subscript: q("subscript"),
   347→      ul: q("insertUnorderedList"),
   348→      ol: q("insertOrderedList"),
   349→      block,
   350→      align: q("justifyCenter")
   351→        ? "center"
   352→        : q("justifyRight")
   353→          ? "right"
   354→          : q("justifyFull")
   355→            ? "full"
   356→            : "left",
   357→      fontName,
   358→      fontSize: fontSizePt,
   359→      link: !!elem?.closest("a"),
   360→    })
   361→  }, [])
   362→
   363→  React.useEffect(() => {
   364→    const onSel = () => {
   365→      const el = pageRef.current
   366→      if (!el) return
   367→      const sel = window.getSelection()
   368→      if (!sel || !sel.anchorNode || !el.contains(sel.anchorNode)) return
   369→      if (sel.rangeCount > 0) savedRangeRef.current = sel.getRangeAt(0).cloneRange()
   370→      refreshFmt()
   371→      setTableInfo(describeTableAt(el))
   372→      const ctx = getTableContext(el)
   373→      setActiveTableEl(ctx ? ctx.table : null)
   374→      const now = Date.now()
   375→      if (now - lastCursorEmitRef.current > 140) {
   376→        lastCursorEmitRef.current = now
   377→        const off = selectionOffsets(el)
   378→        if (off) emitCursor(off.start, off.end)
   379→      }
   380→
   381→      // comment bubble: show above a non-collapsed selection with text
   382→      const inside = sel.rangeCount > 0 && el.contains(sel.focusNode)
   383→      if (inside && !sel.isCollapsed && !pendingQuoteRef.current) {
   384→        const text = sel.toString().replace(/\s+/g, " ").trim()
   385→        if (text) {
   386→          const rect = sel.getRangeAt(0).getBoundingClientRect()
   387→          setBubble({ x: rect.left + rect.width / 2, y: rect.top - 38 })
   388→        } else {
   389→          setBubble(null)
   390→        }
   391→      } else {
   392→        setBubble(null)
   393→      }
   394→    }
   395→    document.addEventListener("selectionchange", onSel)
   396→    return () => document.removeEventListener("selectionchange", onSel)
   397→  }, [refreshFmt, emitCursor])
   398→
   399→  // hide the comment bubble whenever any scroll happens (position would drift)
   400→  React.useEffect(() => {
   401→    const onScroll = () => setBubble(null)
   402→    window.addEventListener("scroll", onScroll, true)
   403→    return () => window.removeEventListener("scroll", onScroll, true)
   404→  }, [])
   405→
   406→  /* ---------------- editing commands ---------------- */
   407→  /** Focus the editor and make sure the last valid selection is active — so
   408→   *  toolbar clicks (which blur the editor) never lose the user's selection. */
   409→  const ensureSelection = React.useCallback(() => {
   410→    const el = pageRef.current
   411→    if (!el) return
   412→    el.focus()
   413→    const sel = window.getSelection()
   414→    const inside = !!sel && sel.rangeCount > 0 && !!sel.anchorNode && el.contains(sel.anchorNode)
   415→    if (!inside && savedRangeRef.current) {
   416→      try {
   417→        sel?.removeAllRanges()
   418→        sel?.addRange(savedRangeRef.current.cloneRange())
   419→      } catch {
   420→        // saved range no longer valid
   421→      }
   422→    }
   423→  }, [])
   424→
   425→  const exec = React.useCallback(
   426→    (cmd: string, val?: string) => {
   427→      const el = pageRef.current
   428→      if (!el) return
   429→      ensureSelection()
   430→      try {
   431→        document.execCommand(cmd, false, val)
   432→      } catch {
   433→        // command unsupported — ignore
   434→      }
   435→      handleInput()
   436→      refreshFmt()
   437→    },
   438→    [handleInput, refreshFmt, ensureSelection]
   439→  )
   440→
   441→  const applyFontSize = React.useCallback(
   442→    (pt: number) => {
   443→      const el = pageRef.current
   444→      if (!el) return
   445→      ensureSelection()
   446→      document.execCommand("fontSize", false, "7")
   447→      el.querySelectorAll('font[size="7"]').forEach((f) => {
   448→        const span = document.createElement("span")
   449→        span.style.fontSize = `${pt}pt`
   450→        span.innerHTML = f.innerHTML
   451→        f.replaceWith(span)
   452→      })
   453→      handleInput()
   454→      refreshFmt()
   455→    },
   456→    [handleInput, refreshFmt]
   457→  )
   458→
   459→  const applyLineSpacing = React.useCallback(
   460→    (lh: number) => {
   461→      const el = pageRef.current
   462→      if (!el) return
   463→      el.focus()
   464→      const blocks = selectedBlocks(el)
   465→      blocks.forEach((b) => {
   466→        b.style.lineHeight = String(lh)
   467→      })
   468→      handleInput()
   469→    },
   470→    [handleInput]
   471→  )
   472→
   473→  const applyAlignment = React.useCallback(
   474→    (a: "left" | "center" | "right" | "full") => {
   475→      const cmd =
   476→        a === "center" ? "justifyCenter" : a === "right" ? "justifyRight" : a === "full" ? "justifyFull" : "justifyLeft"
   477→      exec(cmd)
   478→    },
   479→    [exec]
   480→  )
   481→
   482→  const clearFormatting = React.useCallback(() => {
   483→    exec("removeFormat")
   484→    exec("formatBlock", "p")
   485→  }, [exec])
   486→
   487→  const focusEditor = React.useCallback(() => {
   488→    const el = pageRef.current
   489→    if (!el) return
   490→    el.focus()
   491→    const sel = window.getSelection()
   492→    const range = document.createRange()
   493→    range.selectNodeContents(el)
   494→    range.collapse(false)
   495→    sel?.removeAllRanges()
   496→    sel?.addRange(range)
   497→  }, [])
   498→
   499→  const onTitleChange = React.useCallback(
   500→    (t: string) => {
   501→      setTitle(t)
   502→      latestTitleRef.current = t
   503→      scheduleSave()
   504→      scheduleBroadcast()
   505→    },
   506→    [scheduleSave, scheduleBroadcast]
   507→  )
   508→
   509→  /* ---------------- insert helpers ---------------- */
   510→  const insertLink = React.useCallback(
   511→    (url: string, text: string) => {
   512→      const el = pageRef.current
   513→      if (!el) return
   514→      ensureSelection()
   515→      const sel = window.getSelection()
   516→      const collapsed = !sel || sel.isCollapsed || !el.contains(sel.anchorNode)
   517→      if (collapsed) {
   518→        const safeUrl = url.replace(/"/g, "%22")
   519→        document.execCommand(
   520→          "insertHTML",
   521→          false,
   522→          `<a href="${safeUrl}">${escapeHtml(text || url)}</a>`
   523→        )
   524→      } else {
   525→        document.execCommand("createLink", false, url)
   526→      }
   527→      handleInput()
   528→      refreshFmt()
   529→    },
   530→    [handleInput, refreshFmt, ensureSelection]
   531→  )
   532→
   533→  const insertImage = React.useCallback(
   534→    (src: string, alt: string) => {
   535→      const el = pageRef.current
   536→      if (!el) return
   537→      ensureSelection()
   538→      const safeSrc = src.replace(/"/g, "%22")
   539→      document.execCommand(
   540→        "insertHTML",
   541→        false,
   542→        `<img src="${safeSrc}" alt="${escapeHtml(alt)}" style="max-width:100%">`
   543→      )
   544→      handleInput()
   545→    },
   546→    [handleInput]
   547→  )
   548→
   549→  const insertHtmlAtCursor = React.useCallback(
   550→    (html: string) => {
   551→      const el = pageRef.current

  1119→        e.preventDefault()
  1120→        printDoc()
  1121→      } else if (k === "t" && e.shiftKey) {
  1122→        e.preventDefault()
  1123→        setDialog("table")
  1124→      } else if (k === "\\") {
  1125→        e.preventDefault()
  1126→        clearFormatting()
  1127→      } else if (k === "m" && e.altKey) {
  1128→        e.preventDefault()
  1129→        openCommentComposer()
  1130→      } else if (k === "a" && e.altKey) {
  1131→        // ⌥⌘A — AI polish on the selection (or whole document)
  1132→        e.preventDefault()
  1133→        openAiTools()
  1134→        setDialog("aitools")
  1135→      } else if (k === "escape") {
  1136→        setBubble(null)
  1137→      }
  1138→    }

  1189→      el.classList.add("outline-flash")

  1208→        range = savedRangeRef.current

  1211→        el.insertAdjacentHTML("beforeend", escapeHtml(emoji))
  1212→      } else {

  1214→          range.deleteContents()

  1219→          sel?.removeAllRanges()
  1220→          sel?.addRange(range)
  1221→          savedRangeRef.current = range.cloneRange()

  1223→          el.insertAdjacentHTML("beforeend", escapeHtml(emoji))
