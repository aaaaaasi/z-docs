    40→function titleFor(filter: string, folders: FolderDTO[], activeFolderId: string | null) {
    41→  if (filter === "trash") return "Trash"
    42→  if (filter === "starred") return "Starred"
    43→  if (filter === "folder") {
    44→    const f = folders.find((x) => x.id === activeFolderId)
    45→    return f ? f.name : "Folder"
    46→  }
    47→  return "Recent documents"
    48→}

   335→                selectionActive={selectionActive}

   338→                onStar={() => toggleStar(doc.id)}
   339→                onRename={() => { setRenaming(doc); setRenameValue(doc.title) }}

   341→                onRestore={() => { void setTrashed(doc.id, false); toast({ title: t("Restored"), description: doc.title }) }}

   343→                onDuplicate={() => { void duplicateDoc(doc.id); toast({ title: t("Copy created") }) }}
   344→                onMoveToFolder={(folderId) => {
   345→                  void moveToFolder(doc.id, folderId)
   346→                  const f = folders.find((x) => x.id === folderId)
   347→                  toast({ title: f ? t("Moved to “{name}”", { name: f.name }) : t("Removed from folder") })

   358→          <DialogHeader>
   359→            <DialogTitle>{t("Rename document")}</DialogTitle>

   376→      {/* Delete forever confirm */}

   381→            <AlertDialogDescription>

   385→          <AlertDialogFooter>
   386→            <AlertDialogCancel>{t("Cancel")}</AlertDialogCancel>

   388→              className="bg-destructive text-white hover:bg-destructive/90"
   389→              onClick={confirmDelete}

   391→              {t("Delete forever")}

   393→          </AlertDialogFooter>
   394→        </AlertDialogContent>
   395→      </AlertDialog>
   396→      {/* Batch delete-forever confirm (mirrors the single-doc flow) */}

   398→        <AlertDialogContent>
   399→          <AlertDialogHeader>
   400→            <AlertDialogTitle>{t("Delete {n} documents forever?", { n: selection.length })}</AlertDialogTitle>
   401→            <AlertDialogDescription>
   402→              {t("This can’t be undone. The selected documents will be permanently removed.")}
   403→            </AlertDialogDescription>
   404→          </AlertDialogHeader>
   405→          <AlertDialogFooter>
   406→            <AlertDialogCancel>{t("Cancel")}</AlertDialogCancel>
   407→            <AlertDialogAction
   408→              className="bg-destructive text-white hover:bg-destructive/90"
   409→              onClick={() => {
   410→                setConfirmBatchDelete(false)
   411→                void runBatch("deleteForever")
   412→              }}
   413→            >
   414→              {t("Delete forever")}
   415→            </AlertDialogAction>
   416→          </AlertDialogFooter>
   417→        </AlertDialogContent>
   418→      </AlertDialog>
   419→
   420→      {/* Bulk tag picker: pick/create tags, applied additively to the selection */}
   421→      <BulkTagDialog
   422→        open={bulkTagOpen}
   423→        onOpenChange={setBulkTagOpen}
   424→        count={selection.length}
   425→        busy={busy === "tag"}
   426→        onApply={(tagIds) => {
   427→          setBulkTagOpen(false)
   428→          void runBatch("tag", undefined, undefined, tagIds)
   429→        }}
   430→      />
   431→
   432→      {/* Bulk rename: find & replace or numbered patterns, live preview */}
   433→      <BulkRenameDialog
   434→        open={bulkRenameOpen}
   435→        onOpenChange={setBulkRenameOpen}
   436→        docs={visible
   437→          .filter((d) => selection.includes(d.id))
   438→          .map((d) => ({ id: d.id, title: d.title }))}
   439→        busy={busy === "rename"}
   440→        onApply={(pairs) => {
   441→          setBulkRenameOpen(false)
   442→          void applyBulkRename(pairs)
   443→        }}
   444→      />
   445→
   446→      {/* Floating bulk-action bar (portaled to body: the animate-view-in wrapper
   447→          keeps a persistent transform, which would break viewport-fixed children) */}
   448→      {selectionActive &&
   449→        !loading &&
   450→        createPortal(
   451→          <BatchActionBar
   452→            count={selection.length}
   453→            inTrash={filter === "trash"}
   454→            folders={folders}
   455→            busy={busy}
   456→            onRun={runBatch}
   457→            onOpenTags={() => setBulkTagOpen(true)}
   458→            onOpenRename={() => setBulkRenameOpen(true)}
   459→            onClose={clearSelection}
   460→          />,
   461→          document.body
   462→        )}
   463→    </section>
   464→  )
   465→}

   478→function MoveToSubmenu({

   484→  docFolderId: string | null | undefined

   487→  const { t } = useI18n()

   510→/** "Add tags" submenu: checkbox per tag + inline "New tag…" creation (applied to this doc) */
   511→function AddTagsSubmenu({ doc }: { doc: DocumentMeta }) {
   512→  const tags = useDocsStore((s) => s.tags)
   513→  const setDocTags = useDocsStore((s) => s.setDocTags)
   514→  const createTag = useDocsStore((s) => s.createTag)
   515→  const { toast } = useToast()
   516→  const { t } = useI18n()
   517→  const [creating, setCreating] = React.useState(false)
   518→  const [name, setName] = React.useState("")
   519→  const [color, setColor] = React.useState<string>(DEFAULT_TAG_COLOR)
   520→  const docTagIds = new Set((doc.tags ?? []).map((tg) => tg.id))
   521→
   522→  const toggleTag = (tagId: string) => {
   523→    const ids = (doc.tags ?? []).map((tg) => tg.id)
   524→    void setDocTags(doc.id, ids.includes(tagId) ? ids.filter((i) => i !== tagId) : [...ids, tagId])
   525→  }
   526→
   527→  const confirmCreate = async () => {
   528→    const trimmed = name.trim()
   529→    if (!trimmed) return
   530→    const tag = await createTag(trimmed, color)
   531→    if (tag) {
   532→      const ids = (doc.tags ?? []).map((tg) => tg.id)
   533→      if (!ids.includes(tag.id)) void setDocTags(doc.id, [...ids, tag.id])
   534→      toast({ title: t("Tag created"), description: t("“{name}” added to this document.", { name: tag.name }) })
   535→      setCreating(false)
   536→      setName("")
   537→    } else {
   538→      toast({ title: t("Couldn’t create tag"), description: t("Names must be unique."), variant: "destructive" })
   539→    }
   540→  }
   541→
   542→  return (
   543→    <DropdownMenuSub>
   544→      <DropdownMenuSubTrigger className="gap-2">
   545→        <Tag className="h-4 w-4" /> {t("Add tags")}
   546→      </DropdownMenuSubTrigger>
   547→      <DropdownMenuSubContent className="w-52">
   548→        {tags.map((tag) => (
   549→          <DropdownMenuCheckboxItem
   550→            key={tag.id}
   551→            checked={docTagIds.has(tag.id)}
   552→            onCheckedChange={() => toggleTag(tag.id)}
   553→            onSelect={(e) => e.preventDefault()}
   554→          >
   555→            <span
   556→              aria-hidden="true"
   557→              className="h-2 w-2 shrink-0 rounded-full"
   558→              style={{ backgroundColor: tag.color }}
   559→            />
   560→            <span className="truncate">{tag.name}</span>
   561→          </DropdownMenuCheckboxItem>
   562→        ))}
   563→        {tags.length > 0 && <DropdownMenuSeparator />}
   564→        {creating ? (
   565→          <div
   566→            className="flex flex-col gap-1.5 px-2 py-1.5"
   567→            onBlur={(e) => {
   568→              // cancel only when focus leaves the whole creation form (palette clicks stay inside)
   569→              if (!e.currentTarget.contains(e.relatedTarget as Node | null)) setCreating(false)
   570→            }}
   571→          >
   572→            <div className="flex items-center gap-2">
   573→              <span
   574→                aria-hidden="true"
   575→                className="h-2 w-2 shrink-0 rounded-full"
   576→                style={{ backgroundColor: color }}
   577→              />
   578→              <Input
   579→                autoFocus
   580→                value={name}
   581→                onChange={(e) => setName(e.target.value)}
   582→                onKeyDown={(e) => {
   583→                  // stop the Radix menu typeahead from stealing printable keys / arrows
   584→                  e.stopPropagation()
   585→                  if (e.key === "Enter") void confirmCreate()
   586→                  if (e.key === "Escape") setCreating(false)
   587→                }}
   588→                placeholder={t("Tag name")}
   589→                aria-label={t("New tag name")}
   590→                className="h-8 rounded-md text-[13px]"
   591→                maxLength={24}
   592→              />
   593→            </div>
   594→            <div className="pl-6">
   595→              <TagColorPalette value={color} onChange={setColor} />
   596→            </div>
   597→          </div>
   598→        ) : (
   599→          <DropdownMenuItem
   600→            onSelect={(e) => {
   601→              e.preventDefault()

   603→              setName("")
   604→            }}
   605→          >
   606→            <Plus className="h-4 w-4" /> {t("New tag…")}
   607→          </DropdownMenuItem>
   608→        )}
   609→        {tags.length === 0 && !creating && (
   610→          <p className="px-2 py-1.5 text-[11.5px] leading-relaxed text-muted-foreground/80">
   611→            {t("No tags yet. Create one to label documents.")}

   615→    </DropdownMenuSub>

  1147→/**
  1148→ * Bulk rename dialog — Google-Drive-add-on parity: find & replace inside
  1149→ * names, or numbered patterns with {n} (sequence) / {title} (current name)
  1150→ * placeholders. A live preview lists every resulting name; only documents
  1151→ * whose name actually changes are PATCHed on apply.
  1152→ */
  1153→function BulkRenameDialog({
  1154→  open,
  1155→  onOpenChange,
  1156→  docs,
  1157→  busy,
  1158→  onApply,
  1159→}: {
  1160→  open: boolean
  1161→  onOpenChange: (open: boolean) => void
  1162→  docs: { id: string; title: string }[]
  1163→  busy: boolean
  1164→  onApply: (pairs: { id: string; title: string }[]) => void
  1165→}) {
  1166→  const { t } = useI18n()
  1167→  const [mode, setMode] = React.useState<"replace" | "number">("replace")
  1168→  const [find, setFind] = React.useState("")
  1169→  const [repl, setRepl] = React.useState("")
  1170→  const [pattern, setPattern] = React.useState("")
  1171→  const [start, setStart] = React.useState(1)
  1172→
  1173→  // reset the draft whenever the dialog (re)opens
  1174→  React.useEffect(() => {
  1175→    if (open) {
  1176→      setMode("replace")
  1177→      setFind("")
  1178→      setRepl("")
  1179→      setPattern("")
  1180→      setStart(1)
  1181→    }
  1182→  }, [open])
  1183→
  1184→  const rows = React.useMemo(() => {
  1185→    const n0 = Number.isFinite(start) ? Math.trunc(start) : 1
  1186→    return docs.map((d, i) => {
  1187→      let next = d.title
  1188→      if (mode === "replace") {
  1189→        if (find) next = d.title.split(find).join(repl)
  1190→      } else {
  1191→        const eff = pattern.trim() || "{title} {n}"
  1192→        next = eff.replace(/\{title\}/g, d.title).replace(/\{n\}/g, String(n0 + i))
  1193→      }
  1194→      return { id: d.id, title: d.title, next: next.trim().slice(0, 150) }
  1195→    })
  1196→  }, [docs, mode, find, repl, pattern, start])
  1197→
  1198→  const changed = rows.filter((r) => r.next !== r.title && r.next !== "")
  1199→  const preview = rows.slice(0, 6)
  1200→  const more = rows.length - preview.length
  1201→
  1202→  return (
  1203→    <Dialog open={open} onOpenChange={onOpenChange}>
  1204→      <DialogContent className="sm:max-w-md">
  1205→        <DialogHeader>
  1206→          <DialogTitle>{t("Rename {n} documents", { n: docs.length })}</DialogTitle>
  1207→        </DialogHeader>
  1208→
  1209→        <Tabs value={mode} onValueChange={(v) => setMode(v as "replace" | "number")}>
  1210→          <TabsList className="w-full">
  1211→            <TabsTrigger value="replace" className="flex-1">{t("Find & replace")}</TabsTrigger>
  1212→            <TabsTrigger value="number" className="flex-1">{t("Add numbering")}</TabsTrigger>
  1213→          </TabsList>
  1214→
  1215→          <TabsContent value="replace" className="space-y-3 pt-2">
  1216→            <div className="space-y-2">
  1217→              <label htmlFor="bulk-find" className="text-[13px] font-medium">{t("Find")}</label>
  1218→              <Input
  1219→                id="bulk-find"
  1220→                value={find}
  1221→                maxLength={100}
  1222→                autoFocus
  1223→                onChange={(e) => setFind(e.target.value)}
  1224→                placeholder={t("Text to find in the names…")}
  1225→              />
  1226→            </div>
  1227→            <div className="space-y-2">
  1228→              <label htmlFor="bulk-repl" className="text-[13px] font-medium">{t("Replace with")}</label>
  1229→              <Input
  1230→                id="bulk-repl"
  1231→                value={repl}
  1232→                maxLength={150}
  1233→                onChange={(e) => setRepl(e.target.value)}
  1234→                placeholder={t("Replacement text…")}
  1235→              />
  1236→              <p className="text-xs text-muted-foreground">{t("Case-sensitive. Empty replacement deletes the text.")}</p>
  1237→            </div>
  1238→          </TabsContent>
  1239→
  1240→          <TabsContent value="number" className="space-y-3 pt-2">
  1241→            <div className="space-y-2">
  1242→              <label htmlFor="bulk-pattern" className="text-[13px] font-medium">{t("Name pattern")}</label>
  1243→              <Input
  1244→                id="bulk-pattern"
  1245→                value={pattern}
  1246→                maxLength={120}
  1247→                onChange={(e) => setPattern(e.target.value)}
  1248→                placeholder={t("e.g. Report {n}")}
  1249→              />
  1250→              <p className="text-xs text-muted-foreground">
  1251→                {t("{n} = sequence number, {title} = current name. Leave empty to use “{title} {n}”.")}
  1252→              </p>
  1253→            </div>
  1254→            <div className="space-y-2">
  1255→              <label htmlFor="bulk-start" className="text-[13px] font-medium">{t("Start number")}</label>
  1256→              <Input
  1257→                id="bulk-start"
  1258→                type="number"
  1259→                min={0}
  1260→                step={1}
  1261→                value={start}
  1262→                onChange={(e) => setStart(Number(e.target.value))}
  1263→                className="w-28"
  1264→              />
  1265→            </div>
  1266→          </TabsContent>
  1267→        </Tabs>
  1268→
  1269→        {/* live preview — every resulting name before anything is applied */}
  1270→        <div className="rounded-lg border">
  1271→          <p className="border-b px-3 py-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
  1272→            {t("Preview")}
  1273→          </p>
  1274→          <div className="max-h-44 overflow-y-auto px-3 py-2" role="list" aria-label={t("Preview")}>
  1275→            {preview.map((r) => {
  1276→              const same = r.next === r.title || r.next === ""
  1277→              return (
  1278→                <div key={r.id} className="flex items-center gap-2 py-1 text-[13px]" role="listitem">
  1279→                  <span className="min-w-0 flex-1 truncate text-muted-foreground" title={r.title}>
  1280→                    {r.title}
  1281→                  </span>
  1282→                  <ArrowRight className="h-3.5 w-3.5 shrink-0 text-muted-foreground/60" aria-hidden />
  1283→                  <span
  1284→                    className={cn("min-w-0 flex-1 truncate", same ? "italic text-muted-foreground/70" : "font-medium")}
  1285→                    title={r.next}
  1286→                  >
  1287→                    {same ? t("No change") : r.next}
  1288→                  </span>
  1289→                </div>
  1290→              )
  1291→            })}
  1292→            {more > 0 && (
  1293→              <p className="py-1 pl-1 text-xs text-muted-foreground">+{more}</p>
  1294→            )}
  1295→          </div>
  1296→        </div>
  1297→
  1298→        <DialogFooter className="gap-2 sm:gap-0">
  1299→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>
  1300→            {t("Cancel")}
  1301→          </Button>
  1302→          <Button onClick={() => onApply(changed)} disabled={busy || changed.length === 0} className="gap-1.5">
  1303→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Pencil className="h-4 w-4" />}
  1304→            {changed.length > 0
  1305→              ? t("Rename {n} documents", { n: changed.length })
  1306→              : t("Rename")}
  1307→          </Button>
  1308→        </DialogFooter>
  1309→      </DialogContent>
  1310→    </Dialog>
  1311→  )
  1312→}
