    36→function titleFor(filter: string, folders: FolderDTO[], activeFolderId: string | null) {
    37→  if (filter === "trash") return "Trash"
    38→  if (filter === "starred") return "Starred"
    39→  if (filter === "folder") {
    40→    const f = folders.find((x) => x.id === activeFolderId)
    41→    return f ? f.name : "Folder"
    42→  }
    43→  return "Recent documents"
    44→}

   310→/** "Add tags" submenu: checkbox per tag + inline "New tag…" creation (applied to this doc) */
   311→function AddTagsSubmenu({ doc }: { doc: DocumentMeta }) {
   312→  const tags = useDocsStore((s) => s.tags)
   313→  const setDocTags = useDocsStore((s) => s.setDocTags)
   314→  const createTag = useDocsStore((s) => s.createTag)
   315→  const { toast } = useToast()
   316→  const [creating, setCreating] = React.useState(false)
   317→  const [name, setName] = React.useState("")
   318→  const [color, setColor] = React.useState<string>(DEFAULT_TAG_COLOR)
   319→  const docTagIds = new Set((doc.tags ?? []).map((t) => t.id))
   320→
   321→  const toggleTag = (tagId: string) => {
   322→    const ids = (doc.tags ?? []).map((t) => t.id)
   323→    void setDocTags(doc.id, ids.includes(tagId) ? ids.filter((i) => i !== tagId) : [...ids, tagId])
   324→  }
   325→
   326→  const confirmCreate = async () => {
   327→    const trimmed = name.trim()
   328→    if (!trimmed) return
   329→    const tag = await createTag(trimmed, color)
   330→    if (tag) {
   331→      const ids = (doc.tags ?? []).map((t) => t.id)
   332→      if (!ids.includes(tag.id)) void setDocTags(doc.id, [...ids, tag.id])
   333→      toast({ title: "Tag created", description: `“${tag.name}” added to this document.` })
   334→      setCreating(false)
   335→      setName("")
   336→    } else {
   337→      toast({ title: "Couldn’t create tag", description: "Names must be unique.", variant: "destructive" })
   338→    }
   339→  }
   340→
   341→  return (
   342→    <DropdownMenuSub>
   343→      <DropdownMenuSubTrigger className="gap-2">
   344→        <Tag className="h-4 w-4" /> Add tags
   345→      </DropdownMenuSubTrigger>
   346→      <DropdownMenuSubContent className="w-52">
   347→        {tags.map((t) => (
   348→          <DropdownMenuCheckboxItem
   349→            key={t.id}
   350→            checked={docTagIds.has(t.id)}
   351→            onCheckedChange={() => toggleTag(t.id)}
   352→            onSelect={(e) => e.preventDefault()}
   353→          >
   354→            <span
   355→              aria-hidden="true"
   356→              className="h-2 w-2 shrink-0 rounded-full"
   357→              style={{ backgroundColor: t.color }}
   358→            />
   359→            <span className="truncate">{t.name}</span>
   360→          </DropdownMenuCheckboxItem>
   361→        ))}
   362→        {tags.length > 0 && <DropdownMenuSeparator />}
   363→        {creating ? (
   364→          <div
   365→            className="flex flex-col gap-1.5 px-2 py-1.5"
   366→            onBlur={(e) => {
   367→              // cancel only when focus leaves the whole creation form (palette clicks stay inside)
   368→              if (!e.currentTarget.contains(e.relatedTarget as Node | null)) setCreating(false)
   369→            }}
   370→          >
   371→            <div className="flex items-center gap-2">
   372→              <span
   373→                aria-hidden="true"
   374→                className="h-2 w-2 shrink-0 rounded-full"
   375→                style={{ backgroundColor: color }}
   376→              />
   377→              <Input
   378→                autoFocus
   379→                value={name}
   380→                onChange={(e) => setName(e.target.value)}
   381→                onKeyDown={(e) => {
   382→                  // stop the Radix menu typeahead from stealing printable keys / arrows
   383→                  e.stopPropagation()
   384→                  if (e.key === "Enter") void confirmCreate()
   385→                  if (e.key === "Escape") setCreating(false)
   386→                }}
   387→                placeholder="Tag name"
   388→                aria-label="New tag name"
   389→                className="h-8 rounded-md text-[13px]"
   390→                maxLength={24}
   391→              />
   392→            </div>
   393→            <div className="pl-6">
   394→              <TagColorPalette value={color} onChange={setColor} />
   395→            </div>
   396→          </div>
   397→        ) : (
   398→          <DropdownMenuItem
   399→            onSelect={(e) => {
   400→              e.preventDefault()
   401→              setCreating(true)
   402→              setName("")
   403→            }}
   404→          >
   405→            <Plus className="h-4 w-4" /> New tag…
   406→          </DropdownMenuItem>
   407→        )}
   408→        {tags.length === 0 && !creating && (
   409→          <p className="px-2 py-1.5 text-[11.5px] leading-relaxed text-muted-foreground/80">
   410→            No tags yet. Create one to label documents.
   411→          </p>
   412→        )}
   413→      </DropdownMenuSubContent>
   414→    </DropdownMenuSub>
   415→  )
   416→}

   468→function DocCard({
   469→  doc,
   470→  index,
   471→  inTrash,
   472→  folders,
   473→  ...actions
   474→}: DocActions & { doc: DocumentMeta; index: number; inTrash: boolean; folders: FolderDTO[] }) {
   475→  const folder = folders.find((f) => f.id === doc.folderId)
   476→  const tagFilter = useDocsStore((s) => s.tagFilter)
   477→  const setTagFilter = useDocsStore((s) => s.setTagFilter)
   478→  const docTags = doc.tags ?? []
   479→  const { attributes: dragAttrs, listeners: dragListeners, setNodeRef: dragRef, isDragging } = useDraggable({
   480→    id: docDragId(doc.id),
   481→    disabled: inTrash,
   482→  })
   483→  return (
   484→    <div
   485→      ref={dragRef}
   486→      {...dragAttrs}
   487→      {...dragListeners}
   488→      role="button"
   489→      tabIndex={0}
   490→      onClick={actions.onOpen}
   491→      onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && actions.onOpen()}
   492→      className={cn(
   493→        "animate-card-in group flex w-[168px] cursor-pointer flex-col rounded-lg p-2 transition-colors outline-none hover:bg-muted/60 focus-visible:bg-muted/60",
   494→        isDragging && "opacity-40"
   495→      )}
   496→      style={{ animationDelay: `${Math.min(index * 40, 320)}ms` }}
   497→      aria-label={`Open ${doc.title}`}
   498→    >

   500→        <DocPreview html={doc.content ?? doc.snippet} width={152} className="mx-auto" />
   501→        {doc.starred && !inTrash && (
   502→          <Star className="absolute right-1.5 top-1.5 h-4 w-4 fill-amber-400 text-amber-400 drop-shadow transition-transform duration-200 group-hover:scale-110" aria-label="Starred" />

   510→      <div className="mt-2 flex items-start gap-1">
   511→        <div className="min-w-0 flex-1">
   512→          <p className="truncate text-sm font-medium leading-tight">{doc.title}</p>
   513→          <p className="mt-0.5 truncate text-xs text-muted-foreground">
   514→            {folder && (
   515→              <span className="mr-1 inline-flex items-center gap-0.5 align-baseline text-[11px]" style={{ color: folder.color }}>
   516→                <Folder className="inline h-3 w-3" /> {folder.name}

   518→            )}
   519→            {relativeTime(doc.updatedAt)}
   520→          </p>
   521→          {docTags.length > 0 && !inTrash && (
   522→            <div className="mt-1.5 flex flex-wrap gap-1">
   523→              {docTags.slice(0, 3).map((t) => (
   524→                <TagChip
   525→                  key={t.id}
   526→                  tag={t}
   527→                  active={tagFilter === t.id}
   528→                  onFilter={() => setTagFilter(tagFilter === t.id ? null : t.id)}
   529→                />
   530→              ))}
   531→              {docTags.length > 3 && (
   532→                <TagOverflowChip count={docTags.length - 3} names={docTags.slice(3).map((t) => t.name).join(", ")} />

   534→            </div>
   535→          )}
   536→        </div>
   537→        <CardMenu doc={doc} inTrash={inTrash} folders={folders} {...actions} />
   538→      </div>
   539→    </div>
   540→  )

   543→function DocRow({
   544→  doc,
   545→  index,
   546→  inTrash,
   547→  folders,
   548→  ...actions
   549→}: DocActions & { doc: DocumentMeta; index: number; inTrash: boolean; folders: FolderDTO[] }) {
   550→  const folder = folders.find((f) => f.id === doc.folderId)
   551→  const tagFilter = useDocsStore((s) => s.tagFilter)
   552→  const setTagFilter = useDocsStore((s) => s.setTagFilter)
   553→  const docTags = doc.tags ?? []
   554→  const { attributes: dragAttrs, listeners: dragListeners, setNodeRef: dragRef, isDragging } = useDraggable({
   555→    id: docDragId(doc.id),
   556→    disabled: inTrash,
   557→  })
   558→  return (
   559→    <div
   560→      ref={dragRef}
   561→      {...dragAttrs}
   562→      {...dragListeners}
   563→      role="button"
   564→      tabIndex={0}
   565→      onClick={actions.onOpen}
   566→      onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && actions.onOpen()}
   567→      className={cn(
   568→        "animate-card-in group flex cursor-pointer items-center gap-3 border-b px-4 py-3 outline-none last:border-b-0 hover:bg-muted/50 focus-visible:bg-muted/50",
   569→        index === 0 && "rounded-t-lg", index % 2 === 1 && "bg-muted/30",
   570→        isDragging && "opacity-40"
   571→      )}
   572→      aria-label={`Open ${doc.title}`}
   573→    >
   574→      <FileText className="h-5 w-5 shrink-0 text-muted-foreground" />
   575→      <div className="min-w-0 flex-1">
   576→        <p className="truncate text-sm font-medium">
   577→          {doc.title}
   578→          {doc.starred && !inTrash && <Star className="ml-1.5 inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   579→        </p>
   580→        <p className="truncate text-xs text-muted-foreground">
   581→          {folder && (
   582→            <span className="mr-1 inline-flex items-center gap-0.5 align-baseline text-[11px]" style={{ color: folder.color }}>
   583→              <Folder className="inline h-3 w-3" /> {folder.name}
   584→            </span>
   585→          )}
   586→          {doc.snippet || "Empty document"}
   587→        </p>
   588→        {docTags.length > 0 && !inTrash && (
   589→          <div className="mt-1 flex flex-wrap gap-1">
   590→            {docTags.slice(0, 3).map((t) => (
   591→              <TagChip
   592→                key={t.id}
   593→                tag={t}
   594→                active={tagFilter === t.id}
   595→                onFilter={() => setTagFilter(tagFilter === t.id ? null : t.id)}
   596→              />
   597→            ))}
   598→            {docTags.length > 3 && (
   599→              <TagOverflowChip count={docTags.length - 3} names={docTags.slice(3).map((t) => t.name).join(", ")} />
   600→            )}
   601→          </div>
   602→        )}
   603→      </div>
   604→      <span className="tnum hidden w-20 shrink-0 text-right text-xs text-muted-foreground sm:block">{doc.wordCount} words</span>
   605→      <span className="tnum hidden w-32 shrink-0 text-right text-xs text-muted-foreground md:block">{relativeTime(doc.updatedAt)}</span>
   606→      <CardMenu doc={doc} inTrash={inTrash} folders={folders} {...actions} />
   607→    </div>
   608→  )
   609→}

   611→function EmptyState({ filter, search, tag }: { filter: string; search: boolean; tag?: TagDTO | null }) {
   612→  const createDoc = useDocsStore((s) => s.createDoc)
   613→  const openDoc = useDocsStore((s) => s.openDoc)
   614→  return (
   615→    <div className="flex flex-col items-center justify-center rounded-lg border border-dashed py-16 text-center">
   616→      {search ? (
   617→        <SearchX className="h-10 w-10 text-muted-foreground/40" />
   618→      ) : tag ? (
   619→        <Tag className="h-10 w-10 text-muted-foreground/40" strokeWidth={1.5} />
   620→      ) : filter === "trash" ? (
   621→        <Trash className="h-10 w-10 text-muted-foreground/40" />
   622→      ) : (
   623→        <FileText className="h-10 w-10 text-muted-foreground/40" />
   624→      )}
   625→      <p className="font-editorial mt-4 text-[15.5px] font-medium italic tracking-tight text-foreground/80">
   626→        {search
   627→          ? "No documents match your search"
   628→          : tag
   629→            ? "No documents with this tag"
   630→            : filter === "trash"
   631→              ? "Trash is empty"
   632→              : filter === "starred"
   633→                ? "No starred documents yet"
   634→                : "No documents yet"}
   635→      </p>
   636→      <p className="mt-1.5 max-w-sm text-xs leading-relaxed text-muted-foreground">
   637→        {search
   638→          ? "Try a different keyword or clear the search field."
   639→          : tag
   640→            ? `No documents in this view carry the “${tag.name}” label.`
   641→            : filter === "trash"
   642→              ? "Deleted documents will appear here before being removed forever."
   643→              : "Pick a template above or start from a blank page to create your first document."}
   644→      </p>
   645→      {!search && !tag && filter === "all" && (
   646→        <Button
   647→          className="mt-4"
   648→          onClick={async () => {
   649→            const id = await createDoc({ templateId: "blank" })
   650→            if (id) openDoc(id)
   651→          }}
   652→        >
   653→          Create a document
   654→        </Button>
   655→      )}
   656→    </div>
   657→  )
   658→}
