    37→function titleFor(filter: string, folders: FolderDTO[], activeFolderId: string | null) {
    38→  if (filter === "trash") return "Trash"
    39→  if (filter === "starred") return "Starred"
    40→  if (filter === "folder") {
    41→    const f = folders.find((x) => x.id === activeFolderId)
    42→    return f ? f.name : "Folder"
    43→  }
    44→  return "Recent documents"
    45→}

   313→/** "Add tags" submenu: checkbox per tag + inline "New tag…" creation (applied to this doc) */
   314→function AddTagsSubmenu({ doc }: { doc: DocumentMeta }) {
   315→  const tags = useDocsStore((s) => s.tags)
   316→  const setDocTags = useDocsStore((s) => s.setDocTags)
   317→  const createTag = useDocsStore((s) => s.createTag)
   318→  const { toast } = useToast()
   319→  const { t } = useI18n()
   320→  const [creating, setCreating] = React.useState(false)
   321→  const [name, setName] = React.useState("")
   322→  const [color, setColor] = React.useState<string>(DEFAULT_TAG_COLOR)
   323→  const docTagIds = new Set((doc.tags ?? []).map((tg) => tg.id))
   324→
   325→  const toggleTag = (tagId: string) => {
   326→    const ids = (doc.tags ?? []).map((tg) => tg.id)
   327→    void setDocTags(doc.id, ids.includes(tagId) ? ids.filter((i) => i !== tagId) : [...ids, tagId])
   328→  }
   329→
   330→  const confirmCreate = async () => {
   331→    const trimmed = name.trim()
   332→    if (!trimmed) return
   333→    const tag = await createTag(trimmed, color)
   334→    if (tag) {
   335→      const ids = (doc.tags ?? []).map((tg) => tg.id)
   336→      if (!ids.includes(tag.id)) void setDocTags(doc.id, [...ids, tag.id])
   337→      toast({ title: t("Tag created"), description: t("“{name}” added to this document.", { name: tag.name }) })
   338→      setCreating(false)
   339→      setName("")
   340→    } else {
   341→      toast({ title: t("Couldn’t create tag"), description: t("Names must be unique."), variant: "destructive" })
   342→    }
   343→  }
   344→
   345→  return (
   346→    <DropdownMenuSub>
   347→      <DropdownMenuSubTrigger className="gap-2">
   348→        <Tag className="h-4 w-4" /> {t("Add tags")}
   349→      </DropdownMenuSubTrigger>
   350→      <DropdownMenuSubContent className="w-52">
   351→        {tags.map((tag) => (
   352→          <DropdownMenuCheckboxItem
   353→            key={tag.id}
   354→            checked={docTagIds.has(tag.id)}
   355→            onCheckedChange={() => toggleTag(tag.id)}
   356→            onSelect={(e) => e.preventDefault()}
   357→          >
   358→            <span
   359→              aria-hidden="true"
   360→              className="h-2 w-2 shrink-0 rounded-full"
   361→              style={{ backgroundColor: tag.color }}
   362→            />
   363→            <span className="truncate">{tag.name}</span>
   364→          </DropdownMenuCheckboxItem>
   365→        ))}
   366→        {tags.length > 0 && <DropdownMenuSeparator />}
   367→        {creating ? (
   368→          <div
   369→            className="flex flex-col gap-1.5 px-2 py-1.5"
   370→            onBlur={(e) => {
   371→              // cancel only when focus leaves the whole creation form (palette clicks stay inside)
   372→              if (!e.currentTarget.contains(e.relatedTarget as Node | null)) setCreating(false)
   373→            }}
   374→          >
   375→            <div className="flex items-center gap-2">
   376→              <span
   377→                aria-hidden="true"
   378→                className="h-2 w-2 shrink-0 rounded-full"
   379→                style={{ backgroundColor: color }}
   380→              />
   381→              <Input
   382→                autoFocus
   383→                value={name}
   384→                onChange={(e) => setName(e.target.value)}
   385→                onKeyDown={(e) => {
   386→                  // stop the Radix menu typeahead from stealing printable keys / arrows
   387→                  e.stopPropagation()
   388→                  if (e.key === "Enter") void confirmCreate()
   389→                  if (e.key === "Escape") setCreating(false)
   390→                }}
   391→                placeholder={t("Tag name")}
   392→                aria-label={t("New tag name")}
   393→                className="h-8 rounded-md text-[13px]"
   394→                maxLength={24}
   395→              />
   396→            </div>
   397→            <div className="pl-6">
   398→              <TagColorPalette value={color} onChange={setColor} />
   399→            </div>
   400→          </div>
   401→        ) : (
   402→          <DropdownMenuItem
   403→            onSelect={(e) => {
   404→              e.preventDefault()
   405→              setCreating(true)
   406→              setName("")
   407→            }}
   408→          >
   409→            <Plus className="h-4 w-4" /> {t("New tag…")}
   410→          </DropdownMenuItem>
   411→        )}
   412→        {tags.length === 0 && !creating && (
   413→          <p className="px-2 py-1.5 text-[11.5px] leading-relaxed text-muted-foreground/80">
   414→            {t("No tags yet. Create one to label documents.")}
   415→          </p>
   416→        )}
   417→      </DropdownMenuSubContent>
   418→    </DropdownMenuSub>
   419→  )
   420→}

   473→function DocCard({
   474→  doc,
   475→  index,
   476→  inTrash,
   477→  folders,
   478→  ...actions
   479→}: DocActions & { doc: DocumentMeta; index: number; inTrash: boolean; folders: FolderDTO[] }) {
   480→  const { t, lang } = useI18n()
   481→  const folder = folders.find((f) => f.id === doc.folderId)
   482→  const tagFilter = useDocsStore((s) => s.tagFilter)
   483→  const setTagFilter = useDocsStore((s) => s.setTagFilter)
   484→  const docTags = doc.tags ?? []
   485→  const { attributes: dragAttrs, listeners: dragListeners, setNodeRef: dragRef, isDragging } = useDraggable({
   486→    id: docDragId(doc.id),
   487→    disabled: inTrash,
   488→  })
   489→  return (
   490→    <div
   491→      ref={dragRef}
   492→      {...dragAttrs}
   493→      {...dragListeners}
   494→      role="button"
   495→      tabIndex={0}
   496→      onClick={actions.onOpen}
   497→      onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && actions.onOpen()}
   498→      className={cn(
   499→        "animate-card-in group flex w-[168px] cursor-pointer flex-col rounded-lg p-2 transition-colors outline-none hover:bg-muted/60 focus-visible:bg-muted/60",
   500→        isDragging && "opacity-40"
   501→      )}
   502→      style={{ animationDelay: `${Math.min(index * 40, 320)}ms` }}
   503→      aria-label={t("Open {title}", { title: doc.title })}
   504→    >
   505→      <div className="relative overflow-hidden rounded-sm border elev-1 transition-[border-color,box-shadow] duration-200 group-hover:border-foreground/25 group-hover:elev-2 group-focus-visible:border-foreground/25 group-focus-visible:elev-2">
   506→        <DocPreview html={doc.content ?? doc.snippet} width={152} className="mx-auto" />
   507→        {doc.starred && !inTrash && (
   508→          <Star className="absolute right-1.5 top-1.5 h-4 w-4 fill-amber-400 text-amber-400 drop-shadow transition-transform duration-200 group-hover:scale-110" aria-label={t("Starred")} />
   509→        )}
   510→        {inTrash && (
   511→          <div className="absolute left-1.5 top-1.5 rounded-[4px] bg-black/55 px-2 py-0.5 text-[10px] font-medium text-white">
   512→            {t("In trash")}
   513→          </div>
   514→        )}
   515→      </div>
   516→      <div className="mt-2 flex items-start gap-1">
   517→        <div className="min-w-0 flex-1">
   518→          <p className="truncate text-sm font-medium leading-tight">{doc.title}</p>
   519→          <p className="mt-0.5 truncate text-xs text-muted-foreground">
   520→            {folder && (
   521→              <span className="mr-1 inline-flex items-center gap-0.5 align-baseline text-[11px]" style={{ color: folder.color }}>
   522→                <Folder className="inline h-3 w-3" /> {folder.name}
   523→              </span>
   524→            )}
   525→            {relativeTime(doc.updatedAt, lang)}
   526→          </p>
   527→          {docTags.length > 0 && !inTrash && (
   528→            <div className="mt-1.5 flex flex-wrap gap-1">
   529→              {docTags.slice(0, 3).map((tg) => (
   530→                <TagChip
   531→                  key={tg.id}
   532→                  tag={tg}
   533→                  active={tagFilter === tg.id}
   534→                  onFilter={() => setTagFilter(tagFilter === tg.id ? null : tg.id)}
   535→                />
   536→              ))}
   537→              {docTags.length > 3 && (
   538→                <TagOverflowChip count={docTags.length - 3} names={docTags.slice(3).map((tg) => tg.name).join(", ")} />
   539→              )}
   540→            </div>
   541→          )}
   542→        </div>
   543→        <CardMenu doc={doc} inTrash={inTrash} folders={folders} {...actions} />
   544→      </div>
   545→    </div>
   546→  )
   547→}

   549→function DocRow({
   550→  doc,
   551→  index,
   552→  inTrash,
   553→  folders,
   554→  ...actions
   555→}: DocActions & { doc: DocumentMeta; index: number; inTrash: boolean; folders: FolderDTO[] }) {
   556→  const { t, lang } = useI18n()
   557→  const folder = folders.find((f) => f.id === doc.folderId)
   558→  const tagFilter = useDocsStore((s) => s.tagFilter)
   559→  const setTagFilter = useDocsStore((s) => s.setTagFilter)
   560→  const docTags = doc.tags ?? []
   561→  const { attributes: dragAttrs, listeners: dragListeners, setNodeRef: dragRef, isDragging } = useDraggable({
   562→    id: docDragId(doc.id),
   563→    disabled: inTrash,
   564→  })
   565→  return (
   566→    <div
   567→      ref={dragRef}
   568→      {...dragAttrs}
   569→      {...dragListeners}
   570→      role="button"
   571→      tabIndex={0}
   572→      onClick={actions.onOpen}
   573→      onKeyDown={(e) => (e.key === "Enter" || e.key === " ") && actions.onOpen()}
   574→      className={cn(
   575→        "animate-card-in group flex cursor-pointer items-center gap-3 border-b px-4 py-3 outline-none last:border-b-0 hover:bg-muted/50 focus-visible:bg-muted/50",
   576→        index === 0 && "rounded-t-lg", index % 2 === 1 && "bg-muted/30",
   577→        isDragging && "opacity-40"
   578→      )}
   579→      aria-label={t("Open {title}", { title: doc.title })}
   580→    >
   581→      <FileText className="h-5 w-5 shrink-0 text-muted-foreground" />
   582→      <div className="min-w-0 flex-1">
   583→        <p className="truncate text-sm font-medium">
   584→          {doc.title}
   585→          {doc.starred && !inTrash && <Star className="ml-1.5 inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   586→        </p>
   587→        <p className="truncate text-xs text-muted-foreground">
   588→          {folder && (
   589→            <span className="mr-1 inline-flex items-center gap-0.5 align-baseline text-[11px]" style={{ color: folder.color }}>
   590→              <Folder className="inline h-3 w-3" /> {folder.name}
   591→            </span>
   592→          )}
   593→          {doc.snippet || t("Empty document")}
   594→        </p>
   595→        {docTags.length > 0 && !inTrash && (
   596→          <div className="mt-1 flex flex-wrap gap-1">
   597→            {docTags.slice(0, 3).map((tg) => (
   598→              <TagChip
   599→                key={tg.id}
   600→                tag={tg}
   601→                active={tagFilter === tg.id}
   602→                onFilter={() => setTagFilter(tagFilter === tg.id ? null : tg.id)}
   603→              />
   604→            ))}
   605→            {docTags.length > 3 && (
   606→              <TagOverflowChip count={docTags.length - 3} names={docTags.slice(3).map((tg) => tg.name).join(", ")} />
   607→            )}
   608→          </div>
   609→        )}
   610→      </div>
   611→      <span className="tnum hidden w-20 shrink-0 text-right text-xs text-muted-foreground sm:block">{t("{n} words", { n: doc.wordCount })}</span>
   612→      <span className="tnum hidden w-32 shrink-0 text-right text-xs text-muted-foreground md:block">{relativeTime(doc.updatedAt, lang)}</span>
   613→      <CardMenu doc={doc} inTrash={inTrash} folders={folders} {...actions} />
   614→    </div>
   615→  )
   616→}

   618→function EmptyState({ filter, search, tag }: { filter: string; search: boolean; tag?: TagDTO | null }) {
   619→  const createDoc = useDocsStore((s) => s.createDoc)
   620→  const openDoc = useDocsStore((s) => s.openDoc)
   621→  const { t } = useI18n()
   622→  return (
   623→    <div className="flex flex-col items-center justify-center rounded-lg border border-dashed py-16 text-center">
   624→      {search ? (
   625→        <SearchX className="h-10 w-10 text-muted-foreground/40" />
   626→      ) : tag ? (
   627→        <Tag className="h-10 w-10 text-muted-foreground/40" strokeWidth={1.5} />
   628→      ) : filter === "trash" ? (
   629→        <Trash className="h-10 w-10 text-muted-foreground/40" />
   630→      ) : (
   631→        <FileText className="h-10 w-10 text-muted-foreground/40" />
   632→      )}
   633→      <p className="font-editorial mt-4 text-[15.5px] font-medium italic tracking-tight text-foreground/80">
   634→        {search
   635→          ? t("No documents match your search")
   636→          : tag
   637→            ? t("No documents with this tag")
   638→            : filter === "trash"
   639→              ? t("Trash is empty")
   640→              : filter === "starred"
   641→                ? t("No starred documents yet")
   642→                : t("No documents yet")}
   643→      </p>
   644→      <p className="mt-1.5 max-w-sm text-xs leading-relaxed text-muted-foreground">
   645→        {search
   646→          ? t("Try a different keyword or clear the search field.")
   647→          : tag
   648→            ? t("No documents in this view carry the “{name}” label.", { name: tag.name })
   649→            : filter === "trash"
   650→              ? t("Deleted documents will appear here before being removed forever.")
   651→              : t("Pick a template above or start from a blank page to create your first document.")}
   652→      </p>
   653→      {!search && !tag && filter === "all" && (
   654→        <Button
   655→          className="mt-4"
   656→          onClick={async () => {
   657→            const id = await createDoc({ templateId: "blank" })
   658→            if (id) openDoc(id)
   659→          }}
   660→        >
   661→          {t("Create a document")}
   662→        </Button>
   663→      )}
   664→    </div>
   665→  )
   666→}
